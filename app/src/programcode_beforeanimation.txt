package com.example.pdrandroutetest.hike

import com.example.pdrandroutetest.data.EkfState
import com.example.pdrandroutetest.data.Estimate

import org.junit.Assert.assertFalse
import org.junit.Assert.assertNotNull
import org.junit.Assert.assertNull
import org.junit.Assert.assertTrue
import org.junit.Test

class HikeTest {

    private fun create300mSampleRoute(): Route {
        val wp1 = Waypoint.fromMetric("Start", 0.0, 0.0, legDescription = "Leg 1 to WP2")
        val wp2 = Waypoint.fromMetric("WP2", 0.0, 300.0, legDescription = "Leg 2 to Finish")
        val wp3 = Waypoint.fromMetric("Finish", 0.0, 600.0)
        return Route.createFromWaypoints("Test Route", listOf(wp1, wp2, wp3))
    }

    @Test
    fun `evaluateLegState checks states in strict priority order`() {
        val route = create300mSampleRoute()
        val hike = Hike(route)

        // 1. LEG_START: when leaving start (> 15m from start, at 20m, distance to target is 280m > 100m)
        val stateLegStart = hike.evaluateLegState(Location(0.0, 20.0), currentTimeMs = 1000L)
        assertEquals(LegState.LEG_START, stateLegStart)

        // Second tick at 0.0, 25.0 -> LEG_START already announced, returns OK
        val stateAfterLegStart = hike.evaluateLegState(Location(0.0, 25.0), currentTimeMs = 2000L)
        assertEquals(LegState.OK, stateAfterLegStart)

        // 2. APPROACHING_TARGET: entering approach window (< 100m from target wp2 at 300m, e.g. at 220m)
        val stateApproaching = hike.evaluateLegState(Location(0.0, 220.0), currentTimeMs = 3000L)
        assertEquals(LegState.APPROACHING_TARGET, stateApproaching)

        // Second tick in approach window -> APPROACHING already announced, returns OK
        val stateAfterApproaching = hike.evaluateLegState(Location(0.0, 225.0), currentTimeMs = 4000L)
        assertEquals(LegState.OK, stateAfterApproaching)

        // 3. TARGET_REACHED: arriving at target (< 15m from wp2 at 300m, e.g. at 295m)
        val stateReached = hike.evaluateLegState(Location(0.0, 295.0), currentTimeMs = 5000L)
        assertEquals(LegState.TARGET_REACHED, stateReached)
    }

    @Test
    fun `relativeBearingCorrection and bearingCorrectionString calculate accurate angle offsets`() {
        val wp1 = Waypoint.fromMetric("Start", 0.0, 0.0)
        val wp2 = Waypoint.fromMetric("WP2", 0.0, 150.0) // Target is directly North (0 degrees)
        val route = Route.createFromWaypoints("Route", listOf(wp1, wp2))
        val hike = Hike(route)

        // Case 1: Hiker heading is 345 deg (15 deg left of North) -> correction should be +15 deg (plus 15 degrees)
        val ekfLeft = EkfState(bearingRad = Estimate(Math.toRadians(345.0), 0.01))
        val corrLeft = hike.relativeBearingCorrection(Location(0.0, 20.0), ekfLeft)
        assertEquals(15.0, corrLeft, 0.001)
        assertEquals("plus 15 degrees", hike.bearingCorrectionString(Location(0.0, 20.0), ekfLeft))

        // Case 2: Hiker heading is 30 deg (30 deg right of North) -> correction should be -30 deg (minus 30 degrees)
        val ekfRight = EkfState(bearingRad = Estimate(Math.toRadians(30.0), 0.01))
        val corrRight = hike.relativeBearingCorrection(Location(0.0, 20.0), ekfRight)
        assertEquals(-30.0, corrRight, 0.001)
        assertEquals("minus 30 degrees", hike.bearingCorrectionString(Location(0.0, 20.0), ekfRight))
    }

    @Test
    fun `short leg suppresses LEG_START if APPROACHING_TARGET triggered first`() {
        val wp1 = Waypoint.fromMetric("Start", 0.0, 0.0)
        val wp2 = Waypoint.fromMetric("WP2", 0.0, 80.0)
        val route = Route.createFromWaypoints("Short Route", listOf(wp1, wp2))
        val hike = Hike(route)

        // At location (0, 20): distance to WP2 is 60m (< 100m approach threshold).
        // APPROACHING_TARGET triggers first.
        val state1 = hike.evaluateLegState(Location(0.0, 20.0), currentTimeMs = 1000L)
        assertEquals(LegState.APPROACHING_TARGET, state1)
        assertTrue(hike.tracker.announcedApproaching)

        // Subsequent update at (0, 30): since APPROACHING_TARGET has occurred, LEG_START is suppressed!
        val state2 = hike.evaluateLegState(Location(0.0, 30.0), currentTimeMs = 2000L)
        assertEquals(LegState.OK, state2)
        assertTrue(hike.tracker.announcedLegStart)
    }

    @Test
    fun `evaluateLegState detects TOO_FAR when passing target`() {
        val wp1 = Waypoint.fromMetric("Start", 0.0, 0.0)
        val wp2 = Waypoint.fromMetric("WP2", 0.0, 100.0)
        val route = Route.createFromWaypoints("Route", listOf(wp1, wp2))
        val hike = Hike(route)

        // Position past target along leg direction (t > 1.05 and distance > 15m)
        val stateTooFar = hike.evaluateLegState(Location(0.0, 125.0), currentTimeMs = 1000L)
        assertEquals(LegState.TOO_FAR, stateTooFar)
    }

    @Test
    fun `OFFCOURSE cool-off logic respects 2s drift and 15s steady thresholds`() {
        val route = create300mSampleRoute()
        val hike = Hike(route)

        // Mark LEG_START as announced
        hike.evaluateLegState(Location(0.0, 20.0), currentTimeMs = 1000L)

        // First OFFCOURSE event at t = 2000ms with 30-deg deviation
        val ekf1 = EkfState(bearingRad = Estimate(Math.toRadians(30.0), 0.01))
        val stateOffCourse1 = hike.evaluateLegState(Location(0.0, 20.0), ekfState = ekf1, currentTimeMs = 2000L)
        assertEquals(LegState.OFFCOURSE, stateOffCourse1)

        // At t = 2500ms (500ms later), same 30-deg deviation -> suppressed by 15s cool-off
        val stateOffCourse2 = hike.evaluateLegState(Location(0.0, 20.0), ekfState = ekf1, currentTimeMs = 2500L)
        assertEquals(LegState.OK, stateOffCourse2)

        // At t = 4500ms (2500ms after first offcourse), drifting further to 40-deg (> last + 2.0) -> triggers 2s cool-off
        val ekf2 = EkfState(bearingRad = Estimate(Math.toRadians(40.0), 0.01))
        val stateOffCourse3 = hike.evaluateLegState(Location(0.0, 20.0), ekfState = ekf2, currentTimeMs = 4500L)
        assertEquals(LegState.OFFCOURSE, stateOffCourse3)
    }

    @Test
    fun `stringify returns appropriate message for each LegState`() {
        val route = create300mSampleRoute()
        val hike = Hike(route)
        val leg = route.legs[0]

        assertNull(hike.stringify(LegState.OK, leg, null))

        val msgLegStart = hike.stringify(LegState.LEG_START, leg, null)
        assertTrue(msgLegStart!!.contains("Starting leg"))

        val msgApproaching = hike.stringify(LegState.APPROACHING_TARGET, leg, null)
        assertTrue(msgApproaching!!.contains("Approaching"))

        val msgReached = hike.stringify(LegState.TARGET_REACHED, leg, null)
        assertTrue(msgReached!!.contains("Reached waypoint"))

        val msgTooFar = hike.stringify(LegState.TOO_FAR, leg, null)
        assertTrue(msgTooFar!!.contains("Warning"))

        val msgHikeEnded = hike.stringify(LegState.HIKE_ENDED, leg, null)
        assertTrue(msgHikeEnded!!.contains("Congratulations"))

        val ekfOffCourse = EkfState(bearingRad = Estimate(Math.toRadians(345.0), 0.01))
        val msgOffCourse = hike.stringify(LegState.OFFCOURSE, leg, ekfOffCourse, Location(0.0, 20.0))
        assertEquals("plus 15 degrees", msgOffCourse)
    }

    @Test
    fun `update handles leg progression and string message generation`() {
        val route = create300mSampleRoute()
        val hike = Hike(route)

        val msg1 = hike.update(Location(0.0, 20.0), 1000L, true)
        assertNotNull(msg1)
        assertTrue(msg1!!.contains("Starting leg"))
        assertEquals(0, hike.currentLegIndex)

        // Reach WP2 (at 300m)
        val msg2 = hike.update(Location(0.0, 295.0), 2000L, true)
        assertNotNull(msg2)
        assertTrue(msg2!!.contains("Reached waypoint"))
        assertEquals(1, hike.currentLegIndex)

        // Reach Finish (at 600m)
        val msg3 = hike.update(Location(0.0, 595.0), 3000L, true)
        assertNotNull(msg3)
        assertTrue(msg3!!.contains("Congratulations"))
        assertTrue(hike.isCompleted)
    }

    @Test
    fun `stopping hike ceases activity but preserves state`() {
        val route = create300mSampleRoute()
        val hike = Hike(route)

        hike.update(Location(0.0, 50.0), 1000L, true)
        assertEquals(1, hike.breadcrumbs.size)

        hike.stop()
        assertFalse(hike.isActive)

        val msg = hike.update(Location(0.0, 60.0), 2000L, true)
        assertNull(msg)
        assertEquals(1, hike.breadcrumbs.size)
    }
}

package com.example.pdrandroutetest.telemetry

import com.example.pdrandroutetest.repository.HikeRepository
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.ExperimentalCoroutinesApi
import kotlinx.coroutines.test.TestDispatcher
import kotlinx.coroutines.test.UnconfinedTestDispatcher
import kotlinx.coroutines.test.resetMain
import kotlinx.coroutines.test.runTest
import kotlinx.coroutines.test.setMain
import org.junit.After
import org.junit.Assert.assertEquals
import org.junit.Before
import org.junit.Test
import kotlin.math.PI

@OptIn(ExperimentalCoroutinesApi::class)
class TelemetryPlaybackEngineTest {

    private lateinit var testDispatcher: TestDispatcher
    private lateinit var engine: TelemetryPlaybackEngine

    @Before
    fun setUp() {
        testDispatcher = UnconfinedTestDispatcher()
        Dispatchers.setMain(testDispatcher)
        HikeRepository.clearBreadcrumbs()
        engine = TelemetryPlaybackEngine()
    }

    @After
    fun tearDown() {
        Dispatchers.resetMain()
        HikeRepository.clearBreadcrumbs()
    }

    private fun createSampleEvents(): List<TelemetryEvent> {
        return listOf(
            TelemetryEvent.GpsFixEvent(1000L, 53.349805, -6.260310, 4.0f, 1000L),
            TelemetryEvent.StepDetectedEvent(1600L, PI / 2.0),
            TelemetryEvent.PredictionTickEvent(2000L, 1.0),
            TelemetryEvent.StepDetectedEvent(2200L, PI / 2.0),
            TelemetryEvent.GpsFixEvent(3000L, 53.349810, -6.260305, 4.0f, 1000L)
        )
    }

    @Test
    fun `stepForward advances single event and updates progress`() = runTest {
        val events = createSampleEvents()
        engine.loadEvents(events)

        assertEquals(0, engine.progress.value.currentIndex)
        assertEquals(5, engine.progress.value.totalEvents)

        engine.stepForward()

        assertEquals(1, engine.progress.value.currentIndex)
        assertEquals(1000L, engine.progress.value.currentTimestampMs)
        assertEquals(1, HikeRepository.getBreadcrumbHistory().size)

        engine.stepForward()

        assertEquals(2, engine.progress.value.currentIndex)
        assertEquals(1600L, engine.progress.value.currentTimestampMs)
    }

    @Test
    fun `stepBack rewinds single event and updates breadcrumbs`() = runTest {
        val events = createSampleEvents()
        engine.loadEvents(events)

        engine.stepForward()
        engine.stepForward()
        engine.stepForward()

        assertEquals(3, engine.progress.value.currentIndex)

        engine.stepBack()

        assertEquals(2, engine.progress.value.currentIndex)
        assertEquals(1600L, engine.progress.value.currentTimestampMs)
    }

    @Test
    fun `stepForward and stepBack at boundaries stay within limits`() = runTest {
        val events = createSampleEvents()
        engine.loadEvents(events)

        // Stepping back at 0 does nothing
        engine.stepBack()
        assertEquals(0, engine.progress.value.currentIndex)

        // Step forward 5 times to reach end
        repeat(5) {
            engine.stepForward()
        }
        assertEquals(5, engine.progress.value.currentIndex)

        // Stepping forward at end does nothing
        engine.stepForward()
        assertEquals(5, engine.progress.value.currentIndex)
    }
}

package com.example.pdrandroutetest.ekf

class EkfPlaybackTest {

    private lateinit var ekf: DiagnosticEkf

    // Change this filename to test any recorded CSV file in app/src/test/resources/telemetry/
    private val recordedFileName = "telemetry/telemetry_20261013_153232.csv"

    @Before
    fun setUp() {
        ekf = DiagnosticEkf()
    }

    private fun loadResourceStream(fileName: String) =
        javaClass.classLoader?.getResourceAsStream(fileName)
            ?: javaClass.getResourceAsStream("/$fileName")
            ?: File("src/test/resources/$fileName").takeIf { it.exists() }?.inputStream()

    private fun replaySession(events: List<TelemetryEvent>): List<EkfState> {
        val history = mutableListOf<EkfState>()
        val sortedEvents = events.sortedBy { it.timestampMs }

        for (event in sortedEvents) {
            when (event) {
                is TelemetryEvent.GpsFixEvent -> {
                    val state = ekf.onGpsFix(
                        lat = event.latitude,
                        lon = event.longitude,
                        accuracyMeters = event.accuracyMeters,
                        activeIntervalMs = event.activeIntervalMs
                    )
                    history.add(state)
                }
                is TelemetryEvent.StepDetectedEvent -> {
                    ekf.onStepDetected(event.headingRawRad, event.timestampMs)
                }
                is TelemetryEvent.PredictionTickEvent -> {
                    val state = ekf.predict(event.dtSec)
                    history.add(state)
                }
            }
        }
        return history
    }

    @Test
    fun `ekf initializes position on first GPS fix`() {
        val initialLat = 53.349805
        val initialLon = -6.260310

        assertFalse("EKF should be uninitialized before initial GPS fix", ekf.isInitialized)

        val events = listOf(
            TelemetryEvent.GpsFixEvent(
                timestampMs = 1000L,
                latitude = initialLat,
                longitude = initialLon,
                accuracyMeters = 5.0f,
                activeIntervalMs = 1000L
            )
        )

        val history = replaySession(events)
        val finalState = history.last()

        assertTrue("EKF must be initialized after GPS fix", finalState.isInitializedWithGps)
        val expectedMetric = CoordinateUtils.latLonToMetric(initialLat, initialLon)
        assertEquals(expectedMetric.x, finalState.position.value.x, 0.001)
        assertEquals(expectedMetric.y, finalState.position.value.y, 0.001)
    }

    @Test
    fun `ekf remains stable during synthetic walk playback`() {
        val events = generateSyntheticStraightWalkEvents(
            startLat = 53.349805,
            startLon = -6.260310,
            durationSec = 60,
            stepIntervalMs = 600L,
            gpsIntervalMs = 5000L
        )

        val history = replaySession(events)
        val finalState = history.last()

        assertTrue("EKF must remain initialized throughout playback", finalState.isInitializedWithGps)

        val stride = finalState.strideLengthMeters.value
        assertTrue("Stride length ($stride m) must remain within valid physical bounds [0.4, 1.3]", stride in 0.4..1.3)

        assertFalse("Position variance cannot be NaN", finalState.position.variance.isNaN())
        assertFalse("Position variance cannot be infinite", finalState.position.variance.isInfinite())
        assertTrue("Position variance must stay bounded", finalState.position.variance < 100.0)

        assertTrue("Step count should be tracked during playback", finalState.stepCount > 0)
        assertTrue("GPS fix count should be accumulated", finalState.gpsFixCount > 0)
    }

    @Test
    fun `ekf replays csv telemetry session deterministically`() {
        val recorder = TelemetryRecorder()

        val startLat = 53.349805
        val startLon = -6.260310
        recorder.recordGpsFix(startLat, startLon, 3.0f, 1000L, 1000L)
        recorder.recordStep(PI / 2.0, 1600L)
        recorder.recordTick(1.0, 2000L)
        recorder.recordStep(PI / 2.0, 2200L)
        recorder.recordGpsFix(startLat + 0.00001, startLon + 0.00001, 3.0f, 1000L, 3000L)

        val csvString = recorder.toCsvString()
        val loadedEvents = TelemetryRecorder.fromCsvString(csvString)

        val ekf1 = DiagnosticEkf()
        val ekf2 = DiagnosticEkf()

        var lastState1: EkfState? = null
        for (event in recorder.getEvents()) {
            when (event) {
                is TelemetryEvent.GpsFixEvent -> lastState1 = ekf1.onGpsFix(event.latitude, event.longitude, event.accuracyMeters, event.activeIntervalMs)
                is TelemetryEvent.StepDetectedEvent -> ekf1.onStepDetected(event.headingRawRad, event.timestampMs)
                is TelemetryEvent.PredictionTickEvent -> lastState1 = ekf1.predict(event.dtSec)
            }
        }

        var lastState2: EkfState? = null
        for (event in loadedEvents) {
            when (event) {
                is TelemetryEvent.GpsFixEvent -> lastState2 = ekf2.onGpsFix(event.latitude, event.longitude, event.accuracyMeters, event.activeIntervalMs)
                is TelemetryEvent.StepDetectedEvent -> ekf2.onStepDetected(event.headingRawRad, event.timestampMs)
                is TelemetryEvent.PredictionTickEvent -> lastState2 = ekf2.predict(event.dtSec)
            }
        }

        assertNotNull(lastState1)
        assertNotNull(lastState2)
        assertEquals(lastState1!!.position.value.x, lastState2!!.position.value.x, 1e-6)
        assertEquals(lastState1.position.value.y, lastState2.position.value.y, 1e-6)
        assertEquals(lastState1.strideLengthMeters.value, lastState2.strideLengthMeters.value, 1e-6)
    }

    @Test
    fun `ekf replays telemetry from test resources`() {
        val stream = loadResourceStream(recordedFileName)
            ?: return // Skip gracefully if resource is not present

        val csvText = stream.bufferedReader().use { it.readText() }
        val events = TelemetryRecorder.fromCsvString(csvText)

        val startTime = System.currentTimeMillis()
        val history = replaySession(events)
        val elapsedMs = System.currentTimeMillis() - startTime
        val finalState = history.last()

        println("Replayed ${events.size} telemetry events from $recordedFileName in ${elapsedMs} ms.")
        println("Final State -> Step Count: ${finalState.stepCount}, GPS Fixes: ${finalState.gpsFixCount}")
        println("Final Position -> E: %.2f m, N: %.2f m, Stride: %.2f m, Bias: %.3f rad".format(
            finalState.position.value.x,
            finalState.position.value.y,
            finalState.strideLengthMeters.value,

        ))

        assertTrue("EKF must be initialized after replay", finalState.isInitializedWithGps)
        assertFalse("Position variance cannot be NaN", finalState.position.variance.isNaN())
        assertTrue("Position variance must remain bounded", finalState.position.variance < 100.0)
    }

    @Test
    fun `replayed EKF state matches live recorded EKF state`() {
        val stream = loadResourceStream(recordedFileName)
            ?: return // Skip gracefully if file is not present

        val csvText = stream.bufferedReader().use { it.readText() }
        val loadedEvents = TelemetryRecorder.fromCsvString(csvText)

        val replayEkf = DiagnosticEkf()
        var comparedCount = 0

        for (event in loadedEvents) {
            val currentState = when (event) {
                is TelemetryEvent.GpsFixEvent -> replayEkf.onGpsFix(event.latitude, event.longitude, event.accuracyMeters, event.activeIntervalMs)
                is TelemetryEvent.StepDetectedEvent -> {
                    replayEkf.onStepDetected(event.headingRawRad, event.timestampMs)
                    replayEkf.getEkfState()
                }
                is TelemetryEvent.PredictionTickEvent -> replayEkf.predict(event.dtSec)
            }

            event.recordedEkfState?.let { liveState ->
                comparedCount++
                assertEquals("Position X mismatch at t=${event.timestampMs}", liveState.position.value.x, currentState.position.value.x, 1e-4)
                assertEquals("Position Y mismatch at t=${event.timestampMs}", liveState.position.value.y, currentState.position.value.y, 1e-4)
                assertEquals("Stride Length mismatch at t=${event.timestampMs}", liveState.strideLengthMeters.value, currentState.strideLengthMeters.value, 1e-4)
                assertEquals("Compass Bias mismatch at t=${event.timestampMs}", liveState.compassBiasRad.value, currentState.compassBiasRad.value, 1e-4)
            }
        }
        println("Verified $comparedCount recorded live EKF state snapshots against replayed EKF state!")
    }

    private fun generateSyntheticStraightWalkEvents(
        startLat: Double,
        startLon: Double,
        durationSec: Int,
        stepIntervalMs: Long,
        gpsIntervalMs: Long
    ): List<TelemetryEvent> {
        val events = mutableListOf<TelemetryEvent>()
        var currentTimeMs = 10000L
        val endTimeMs = currentTimeMs + (durationSec * 1000L)

        val headingEastRad = PI / 2.0
        val startMetric = CoordinateUtils.latLonToMetric(startLat, startLon)

        var lastStepMs = currentTimeMs
        var lastGpsMs = currentTimeMs
        var lastTickMs = currentTimeMs

        events.add(TelemetryEvent.GpsFixEvent(currentTimeMs, startLat, startLon, 4.0f, gpsIntervalMs))

        var distanceWalkedMeters = 0.0
        val strideLengthMeters = 0.75

        while (currentTimeMs < endTimeMs) {
            currentTimeMs += 100L

            if (currentTimeMs - lastStepMs >= stepIntervalMs) {
                lastStepMs = currentTimeMs
                distanceWalkedMeters += strideLengthMeters
                events.add(TelemetryEvent.StepDetectedEvent(currentTimeMs, headingEastRad))
            }

            if (currentTimeMs - lastTickMs >= 1000L) {
                lastTickMs = currentTimeMs
                events.add(TelemetryEvent.PredictionTickEvent(currentTimeMs, 1.0))
            }

            if (currentTimeMs - lastGpsMs >= gpsIntervalMs) {
                lastGpsMs = currentTimeMs
                val currentMetric = Location(x = startMetric.x + distanceWalkedMeters, y = startMetric.y, altitude = startMetric.altitude)
                val (lat, lon) = CoordinateUtils.metricToLatLon(currentMetric)
                events.add(TelemetryEvent.GpsFixEvent(currentTimeMs, lat, lon, 4.0f, gpsIntervalMs))
            }
        }

        return events
    }
}

        val requiredBearing = location.bearingTo(leg.target)
        val hikerBearing = ekfState?.let { Math.toDegrees(it.bearingRad.value) }
            ?: previousLocation?.bearingTo(location)
            ?: requiredBearing

        var rawDiff = requiredBearing - hikerBearing
        rawDiff %= 360.0
        if (rawDiff > 180.0) rawDiff -= 360.0
        if (rawDiff < -180.0) rawDiff += 360.0
        return rawDiff
    }

    fun bearingCorrectionString(location: Location, ekfState: EkfState? = null): String {
        val correction = relativeBearingCorrection(location, ekfState)
        val rounded = abs(round(correction)).toInt()
        return if (correction >= 0) {
            "plus $rounded degrees"
        } else {
            "minus $rounded degrees"
        }
    }

    fun evaluateLegState(
        location: Location,
        ekfState: EkfState? = null,
        currentTimeMs: Long = System.currentTimeMillis()
    ): LegState {
        val leg = currentLeg ?: return LegState.HIKE_ENDED
        if (isCompleted) return LegState.HIKE_ENDED

        // 1. TARGET_REACHED & 2. HIKE_ENDED
        if (!tracker.announcedReached) {
            val distanceToTarget = location.distanceTo(leg.target)
            if (distanceToTarget < leg.target.arrive) {
                tracker.announcedReached = true
                return if (leg.next == null) {
                    tracker.announcedHikeEnded = true
                    LegState.HIKE_ENDED
                } else {
                    LegState.TARGET_REACHED
                }
            }
        }

        // 2. HIKE_ENDED (fallback check if already marked)
        if (!tracker.announcedHikeEnded && leg.next == null && location.distanceTo(leg.target) < leg.target.arrive) {
            tracker.announcedHikeEnded = true
            return LegState.HIKE_ENDED
        }

        // 3. TOO_FAR
        if (!tracker.announcedTooFar) {
            if (isPastTarget(location, leg)) {
                tracker.announcedTooFar = true
                return LegState.TOO_FAR
            }
        }

        // 4. APPROACHING_TARGET
        if (!tracker.announcedApproaching) {
            val distanceToTarget = location.distanceTo(leg.target)
            if (distanceToTarget < leg.target.approach) {
                tracker.announcedApproaching = true
                return LegState.APPROACHING_TARGET
            }
        }

        // 5. LEG_START
        if (!tracker.announcedLegStart) {
            // Do not play LEG_START if any of TARGET_REACHED, APPROACHING_TARGET, HIKE_ENDED, or TOO_FAR have occurred
            if (tracker.announcedReached || tracker.announcedHikeEnded || tracker.announcedTooFar || tracker.announcedApproaching) {
                tracker.announcedLegStart = true
            } else {
                val distanceFromStart = location.distanceTo(leg.start)
                if (distanceFromStart > leg.start.arrive) {
                    tracker.announcedLegStart = true
                    return LegState.LEG_START
                }
            }
        }

        // 6. OFFCOURSE
        val requiredBearing = location.bearingTo(leg.target)
        val hikerBearing = ekfState?.let { Math.toDegrees(it.bearingRad.value) }
            ?: previousLocation?.bearingTo(location)
            ?: requiredBearing

        val bearingOffset = angleDifference(hikerBearing, requiredBearing)
        if (bearingOffset > 10.0) {
            val isDriftingFurther = bearingOffset > (lastOffCourseOffset + 2.0)
            val coolOffMs = if (isDriftingFurther) driftCoolOffSec * 1000L else steadyCoolOffSec * 1000L

            if (currentTimeMs - lastOffCourseTimeMs >= coolOffMs) {
                lastOffCourseTimeMs = currentTimeMs
                lastOffCourseOffset = bearingOffset
                return LegState.OFFCOURSE
            }
        }

        // 7. OK
        return LegState.OK
    }

    private fun isPastTarget(p: Location, leg: Leg): Boolean {
        val abx = leg.target.x - leg.start.x
        val aby = leg.target.y - leg.start.y
        val apx = p.x - leg.start.x
        val apy = p.y - leg.start.y

        val abLenSq = abx * abx + aby * aby
        if (abLenSq == 0.0) return false

        val t = (apx * abx + apy * aby) / abLenSq
        val distToTarget = p.distanceTo(leg.target)

        return t > 1.05 && distToTarget > leg.target.arrive
    }

    private fun angleDifference(a: Double, b: Double): Double {
        var diff = (a - b) % 360.0
        if (diff < -180.0) diff += 360.0
        if (diff > 180.0) diff -= 360.0
        return abs(diff)
    }

    fun stringify(state: LegState, leg: Leg?, ekfState: EkfState?, location: Location? = null): String? {
        if (leg == null) return null
        val targetName = leg.target.name.ifBlank { "Next Waypoint" }
        val legDesc = leg.legDescription ?: ""
        val targetDesc = leg.target.description ?: ""

        return when (state) {
            LegState.TARGET_REACHED -> {
                "Reached waypoint $targetName. Proceeding to next leg."
            }
            LegState.HIKE_ENDED -> {
                "Reached waypoint $targetName. Congratulations! You have completed the hike."
            }
            LegState.TOO_FAR -> {
                "Warning! You have gone past waypoint $targetName."
            }
            LegState.APPROACHING_TARGET -> {
                "Approaching waypoint $targetName. $targetDesc"
            }
            LegState.LEG_START -> {
                "Starting leg towards $targetName. $legDesc"
            }
            LegState.OFFCOURSE -> {
                val loc = location ?: previousLocation ?: leg.start
                bearingCorrectionString(loc, ekfState)
            }
            LegState.OK -> {
                null
            }
        }
    }

    fun update(
        location: Location,
        timestampMs: Long = System.currentTimeMillis(),
        isGpsFix: Boolean = false,
        ekfState: EkfState? = null
    ): String? {
        if (!isActive) return null

        val breadcrumb = Breadcrumb(
            id = breadcrumbIdSeq++,
            location = location,
            timestampMs = timestampMs,
            isGpsFix = isGpsFix,
            ekfState = ekfState
        )
        _breadcrumbs.add(breadcrumb)

        val currentRoute = route
        if (currentRoute == null || currentRoute.legs.isEmpty()) {
            previousLocation = location
            return null
        }

        if (isCompleted) {
            return null
        }

        val leg = currentLeg ?: run {
            isCompleted = true
            return null
        }

        val state = evaluateLegState(location, ekfState, timestampMs)

        if (state == LegState.TARGET_REACHED) {
            val nextLeg = leg.next
            if (nextLeg != null) {
                currentLeg = nextLeg
                currentLegIndex++
                tracker.reset()
            } else {
                isCompleted = true
            }
        } else if (state == LegState.HIKE_ENDED) {
            isCompleted = true
        }

        val msg = stringify(state, leg, ekfState, location)
        previousLocation = location
        return msg
    }
}

package com.example.pdrandroutetest.telemetry

import com.example.pdrandroutetest.ekf.DiagnosticEkf
import com.example.pdrandroutetest.model.Breadcrumb
import com.example.pdrandroutetest.model.Location
import com.example.pdrandroutetest.repository.HikeRepository
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.Job
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch
import kotlinx.coroutines.sync.Mutex
import kotlinx.coroutines.sync.withLock

data class PlaybackProgress(
    val currentIndex: Int = 0,
    val totalEvents: Int = 0,
    val currentTimestampMs: Long = 0L
) {
    val fraction: Float
        get() = if (totalEvents > 0) currentIndex.toFloat() / totalEvents.toFloat() else 0f
}

class TelemetryPlaybackEngine {

    private val _events = MutableStateFlow<List<TelemetryEvent>>(emptyList())
    val events: StateFlow<List<TelemetryEvent>> = _events.asStateFlow()

    private val _isPlaying = MutableStateFlow(false)
    val isPlaying: StateFlow<Boolean> = _isPlaying.asStateFlow()

    private val _isPaused = MutableStateFlow(false)
    val isPaused: StateFlow<Boolean> = _isPaused.asStateFlow()

    private val _speedMultiplier = MutableStateFlow(1.0)
    val speedMultiplier: StateFlow<Double> = _speedMultiplier.asStateFlow()

    private val _progress = MutableStateFlow(PlaybackProgress())
    val progress: StateFlow<PlaybackProgress> = _progress.asStateFlow()

    private var playbackJob: Job? = null
    private var ekfEngine = DiagnosticEkf()
    private var breadcrumbIdSeq = 1L
    private var currentIndex = 0
    private val playbackMutex = Mutex()

    @Synchronized
    fun loadEvents(eventList: List<TelemetryEvent>) {
        stop()
        _events.value = eventList.sortedBy { it.timestampMs }
        _progress.value = PlaybackProgress(0, _events.value.size, 0L)
    }

    @Synchronized
    fun play(scope: CoroutineScope) {
        val eventList = _events.value
        if (eventList.isEmpty()) return

        if (_isPaused.value && playbackJob?.isActive == true) {
            _isPaused.value = false
            _isPlaying.value = true
            return
        }

        playbackJob?.cancel()

        _isPlaying.value = true
        _isPaused.value = false

        if (currentIndex <= 0 || currentIndex >= eventList.size || HikeRepository.activeHike == null) {
            ekfEngine.reset()
            HikeRepository.startHike(HikeRepository.activeRoute.value)
            breadcrumbIdSeq = 1L
            currentIndex = 0
        }

        playbackJob = scope.launch(Dispatchers.Default) {
            var lastEventTimeMs = if (currentIndex > 0 && currentIndex <= eventList.size) {
                eventList[currentIndex - 1].timestampMs
            } else {
                0L
            }

            while (currentIndex < eventList.size) {
                while (_isPaused.value) {
                    delay(100L)
                }

                playbackMutex.withLock {
                    if (currentIndex < eventList.size) {
                        val i = currentIndex
                        val event = eventList[i]

                        if (i > 0 && lastEventTimeMs > 0) {
                            val rawDtMs = (event.timestampMs - lastEventTimeMs).coerceAtLeast(0L)
                            val scaledDelayMs = (rawDtMs / _speedMultiplier.value).toLong().coerceIn(1L, 5000L)
                            delay(scaledDelayMs)
                        }
                        lastEventTimeMs = event.timestampMs

                        processEvent(event)
                        currentIndex++

                        _progress.value = PlaybackProgress(
                            currentIndex = currentIndex,
                            totalEvents = eventList.size,
                            currentTimestampMs = event.timestampMs
                        )
                    }
                }
            }

            _isPlaying.value = false
            _isPaused.value = false
        }
    }

    @Synchronized
    fun pause() {
        if (_isPlaying.value) {
            _isPaused.value = true
        }
    }

    @Synchronized
    fun resume() {
        if (_isPaused.value) {
            _isPaused.value = false
        }
    }

    @Synchronized
    fun stop() {
        playbackJob?.cancel()
        playbackJob = null
        _isPlaying.value = false
        _isPaused.value = false
        currentIndex = 0
        ekfEngine.reset()
        HikeRepository.clearBreadcrumbs()
        _progress.value = PlaybackProgress(0, _events.value.size, 0L)
    }

    suspend fun stepForward() = playbackMutex.withLock {
        val eventList = _events.value
        if (eventList.isEmpty()) return@withLock

        if (_isPlaying.value && !_isPaused.value) {
            _isPaused.value = true
        }

        if (currentIndex >= eventList.size) return@withLock

        if (!_isPlaying.value && !_isPaused.value || HikeRepository.activeHike == null) {
            ekfEngine.reset()
            HikeRepository.startHike(HikeRepository.activeRoute.value)
            breadcrumbIdSeq = 1L
            currentIndex = 0
        }
        _isPaused.value = true

        val event = eventList[currentIndex]
        processEvent(event)
        currentIndex++

        _progress.value = PlaybackProgress(
            currentIndex = currentIndex,
            totalEvents = eventList.size,
            currentTimestampMs = event.timestampMs
        )
    }

    suspend fun stepBack() = playbackMutex.withLock {
        val eventList = _events.value
        if (eventList.isEmpty()) return@withLock

        if (_isPlaying.value && !_isPaused.value) {
            _isPaused.value = true
        }

        if (currentIndex <= 0) return@withLock

        val targetIndex = currentIndex - 1
        _isPaused.value = true

        ekfEngine.reset()
        HikeRepository.startHike(HikeRepository.activeRoute.value)
        breadcrumbIdSeq = 1L
        currentIndex = 0

        for (k in 0 until targetIndex) {
            processEvent(eventList[k])
            currentIndex++
        }

        _progress.value = PlaybackProgress(
            currentIndex = targetIndex,
            totalEvents = eventList.size,
            currentTimestampMs = if (targetIndex > 0) eventList[targetIndex - 1].timestampMs else 0L
        )
    }

    private suspend fun processEvent(event: TelemetryEvent) {
        when (event) {
            is TelemetryEvent.GpsFixEvent -> {
                val ekfState = ekfEngine.onGpsFix(
                    lat = event.latitude,
                    lon = event.longitude,
                    accuracyMeters = event.accuracyMeters,
                    activeIntervalMs = event.activeIntervalMs
                )
                HikeRepository.updateEkfState(ekfState)

                val gpsBreadcrumb = Breadcrumb(
                    id = breadcrumbIdSeq++,
                    location = Location(ekfState.position.value.x, ekfState.position.value.y),
                    timestampMs = ekfState.timestampMs,
                    isGpsFix = true,
                    ekfState = ekfState
                )
                HikeRepository.emitBreadcrumb(gpsBreadcrumb)
            }
            is TelemetryEvent.StepDetectedEvent -> {
                ekfEngine.onStepDetected(event.headingRawRad, event.timestampMs)
            }
            is TelemetryEvent.PredictionTickEvent -> {
                if (ekfEngine.isInitialized) {
                    val ekfState = ekfEngine.predict(event.dtSec)
                    HikeRepository.updateEkfState(ekfState)

                    val pdrBreadcrumb = Breadcrumb(
                        id = breadcrumbIdSeq++,
                        location = Location(ekfState.position.value.x, ekfState.position.value.y),
                        timestampMs = ekfState.timestampMs,
                        isGpsFix = false,
                        ekfState = ekfState
                    )
                    HikeRepository.emitBreadcrumb(pdrBreadcrumb)
                }
            }
        }
    }

    fun setSpeedMultiplier(multiplier: Double) {
        _speedMultiplier.value = multiplier.coerceAtLeast(0.1)
    }
}

package com.example.pdrandroutetest.telemetry

class TelemetryRecorder {
    private val events = mutableListOf<TelemetryEvent>()

    @Synchronized
    fun recordGpsFix(
        latitude: Double,
        longitude: Double,
        accuracyMeters: Float,
        activeIntervalMs: Long,
        timestampMs: Long = System.currentTimeMillis(),
        recordedEkfState: EkfState? = null
    ) {

                latitude = latitude,
                longitude = longitude,
                accuracyMeters = accuracyMeters,
                activeIntervalMs = activeIntervalMs,
                recordedEkfState = recordedEkfState
            )
        )
    }

    @Synchronized
    fun recordStep(
        headingRawRad: Double,
        timestampMs: Long = System.currentTimeMillis(),
        recordedEkfState: EkfState? = null
    ) {
        events.add(
            TelemetryEvent.StepDetectedEvent(
                timestampMs = timestampMs,
                headingRawRad = headingRawRad,
                recordedEkfState = recordedEkfState
            )
        )
    }

    @Synchronized
    fun recordTick(
        dtSec: Double = 1.0,
        timestampMs: Long = System.currentTimeMillis(),
        recordedEkfState: EkfState? = null
    ) {
        events.add(
            TelemetryEvent.PredictionTickEvent(
                timestampMs = timestampMs,
                dtSec = dtSec,
                recordedEkfState = recordedEkfState
            )
        )
    }

    @Synchronized
    fun getEvents(): List<TelemetryEvent> {
        return events.toList()
    }

    @Synchronized
    fun clear() {
        events.clear()
    }

    @Synchronized
    fun toCsvString(): String {
        val sb = StringBuilder()
        sb.append("# TELEMETRY_FORMAT_V2\n")
        events.forEach { event ->
            val ekf = event.recordedEkfState
            val ekfSuffix = if (ekf != null) {
                ",${ekf.position.value.x},${ekf.position.value.y},${ekf.velocity.value.x},${ekf.velocity.value.y},${ekf.strideLengthMeters.value},${ekf.compassBiasRad.value},${ekf.position.variance}"
            } else ""

            when (event) {
                is TelemetryEvent.GpsFixEvent -> {
                    sb.append("GPS,${event.timestampMs},${event.latitude},${event.longitude},${event.accuracyMeters},${event.activeIntervalMs}$ekfSuffix\n")
                }
                is TelemetryEvent.StepDetectedEvent -> {
                    sb.append("STEP,${event.timestampMs},${event.headingRawRad}$ekfSuffix\n")
                }
                is TelemetryEvent.PredictionTickEvent -> {
                    sb.append("TICK,${event.timestampMs},${event.dtSec}$ekfSuffix\n")
                }
            }
        }
        return sb.toString()
    }

    @Synchronized
    fun saveToFile(file: File) {
        file.writeText(toCsvString())
    }

    @Synchronized
    fun saveToTimestampedFile(dir: File): File? {
        if (events.isEmpty()) return null
        if (!dir.exists()) {
            dir.mkdirs()
        }
        val timestamp = java.text.SimpleDateFormat("yyyyMMdd_HHmmss", java.util.Locale.US).format(java.util.Date())
        val file = File(dir, "telemetry_$timestamp.csv")
        saveToFile(file)
        return file
    }

    companion object {
        fun fromCsvString(csvString: String): List<TelemetryEvent> {
            val list = mutableListOf<TelemetryEvent>()
            csvString.lines().forEach { rawLine ->
                val line = rawLine.trim()
                if (line.isEmpty() || line.startsWith("#")) return@forEach

                val tokens = line.split(",").map { it.trim() }
                if (tokens.size > 1) {
                    when (tokens[0]) {
                        "GPS" -> {
                            if (tokens.size >= 6) {
                                val ekfState = parseEkfStateFromTokens(tokens, startIndex = 6)
                                list.add(
                                    TelemetryEvent.GpsFixEvent(
                                        timestampMs = tokens[1].toLong(),
                                        latitude = tokens[2].toDouble(),
                                        longitude = tokens[3].toDouble(),
                                        accuracyMeters = tokens[4].toFloat(),
                                        activeIntervalMs = tokens[5].toLong(),
                                        recordedEkfState = ekfState
                                    )
                                )
                            }
                        }
                        "STEP" -> {
                            if (tokens.size >= 3) {
                                val ekfState = parseEkfStateFromTokens(tokens, startIndex = 3)
                                list.add(
                                    TelemetryEvent.StepDetectedEvent(
                                        timestampMs = tokens[1].toLong(),
                                        headingRawRad = tokens[2].toDouble(),
                                        recordedEkfState = ekfState
                                    )
                                )
                            }
                        }
                        "TICK" -> {
                            if (tokens.size >= 3) {
                                val ekfState = parseEkfStateFromTokens(tokens, startIndex = 3)
                                list.add(
                                    TelemetryEvent.PredictionTickEvent(
                                        timestampMs = tokens[1].toLong(),
                                        dtSec = tokens[2].toDouble(),
                                        recordedEkfState = ekfState
                                    )
                                )
                            }
                        }
                    }
                }
            }
            return list
        }

        private fun parseEkfStateFromTokens(tokens: List<String>, startIndex: Int): EkfState? {
            if (tokens.size < startIndex + 7) return null
            return try {
                val posX = tokens[startIndex].toDouble()
                val posY = tokens[startIndex + 1].toDouble()
                val velX = tokens[startIndex + 2].toDouble()
                val velY = tokens[startIndex + 3].toDouble()
                val stride = tokens[startIndex + 4].toDouble()
                val bias = tokens[startIndex + 5].toDouble()
                val posVar = tokens[startIndex + 6].toDouble()

                EkfState(
                    position = Estimate(Vector2D(posX, posY), posVar),
                    velocity = Estimate(Vector2D(velX, velY), 0.1),
                    strideLengthMeters = Estimate(stride, 0.01),
                    compassBiasRad = Estimate(bias, 0.05),
                    isInitializedWithGps = true
                )
            } catch (e: Exception) {
                null
            }
        }

        fun loadFromFile(file: File): List<TelemetryEvent> {
            return fromCsvString(file.readText())
        }
    }
}

package com.example.pdrandroutetest.sensor

import android.content.Context
import android.hardware.Sensor
import android.hardware.SensorEvent
import android.hardware.SensorEventListener
import android.hardware.SensorManager
import kotlin.math.PI
import kotlin.math.atan2
import kotlin.math.sqrt

data class StepDetectorConfig(
    val stepThreshold: Double = 11.2,
    val minStepIntervalMs: Long = 300L,
    val alpha: Double = 0.15
)

class PdrStepDetector(
    context: Context,
    val config: StepDetectorConfig = StepDetectorConfig(),
    private val onStepListener: (headingRawRad: Double, timestampMs: Long) -> Unit
) : SensorEventListener {

    private val sensorManager = context.getSystemService(Context.SENSOR_SERVICE) as SensorManager
    private val accelSensor: Sensor? = sensorManager.getDefaultSensor(Sensor.TYPE_ACCELEROMETER)
    private val rotationVectorSensor: Sensor? = sensorManager.getDefaultSensor(Sensor.TYPE_ROTATION_VECTOR)
    private val gyroSensor: Sensor? = sensorManager.getDefaultSensor(Sensor.TYPE_GYROSCOPE)

    private var isListening = false

    // Step detection parameters
    private var accelFiltered = 9.81
    private var lastPeakTimestampMs = 0L

    // Heading calculation
    private val rotationMatrix = FloatArray(9)
    private val orientationAngles = FloatArray(3)
    @Volatile
    private var currentHeadingRad: Double = 0.0

    fun start() {
        if (isListening) return
        isListening = true

        accelSensor?.let {
            sensorManager.registerListener(this, it, SensorManager.SENSOR_DELAY_GAME)
        }
        rotationVectorSensor?.let {
            sensorManager.registerListener(this, it, SensorManager.SENSOR_DELAY_GAME)
        } ?: gyroSensor?.let {
            sensorManager.registerListener(this, it, SensorManager.SENSOR_DELAY_GAME)
        }
    }

    fun stop() {
        if (!isListening) return
        isListening = false
        sensorManager.unregisterListener(this)
    }

    override fun onSensorChanged(event: SensorEvent) {
        val eventTimestampMs = if (event.timestamp > 0) event.timestamp / 1_000_000L else System.currentTimeMillis()

        when (event.sensor.type) {
            Sensor.TYPE_ROTATION_VECTOR -> {
                SensorManager.getRotationMatrixFromVector(rotationMatrix, event.values)
                SensorManager.getOrientation(rotationMatrix, orientationAngles)
                // Azimuth angle in radians clockwise from North
                var azimuth = orientationAngles[0].toDouble()
                if (azimuth < 0) azimuth += 2 * PI
                currentHeadingRad = azimuth
            }
            Sensor.TYPE_ACCELEROMETER -> {
                val ax = event.values[0]
                val ay = event.values[1]
                val az = event.values[2]
                val mag = sqrt((ax * ax + ay * ay + az * az).toDouble())

                // Apply low pass filter
                accelFiltered = accelFiltered + config.alpha * (mag - accelFiltered)

                if (accelFiltered > config.stepThreshold && (eventTimestampMs - lastPeakTimestampMs) > config.minStepIntervalMs) {
                    lastPeakTimestampMs = eventTimestampMs
                    onStepListener(currentHeadingRad, eventTimestampMs)
                }
            }
            Sensor.TYPE_GYROSCOPE -> {
                if (rotationVectorSensor == null) {
                    val gx = event.values[0].toDouble()
                    val gy = event.values[1].toDouble()
                    currentHeadingRad += (atan2(gy, gx) * 0.02)
                }
            }
        }
    }

    override fun onAccuracyChanged(sensor: Sensor?, accuracy: Int) {}
}

package com.example.pdrandroutetest.ekf

data class EkfConfig(
    val qDiag: DoubleArray = doubleArrayOf(0.05, 0.05, 0.02, 0.02, 0.0001, 0.0001),
    val initialPosVariance: Double = 100.0,
    val initialVelVariance: Double = 0.1,
    val initialStrideVariance: Double = 0.02,
    val initialBiasVariance: Double = 0.05,
    val minStrideMeters: Double = 0.4,
    val maxStrideMeters: Double = 1.3
) {
    override fun equals(other: Any?): Boolean {
        if (this === other) return true
        if (other !is EkfConfig) return false
        return qDiag.contentEquals(other.qDiag) &&
                initialPosVariance == other.initialPosVariance &&
                initialVelVariance == other.initialVelVariance &&
                initialStrideVariance == other.initialStrideVariance &&
                initialBiasVariance == other.initialBiasVariance &&
                minStrideMeters == other.minStrideMeters &&
                maxStrideMeters == other.maxStrideMeters
    }

    override fun hashCode(): Int {
        var result = qDiag.contentHashCode()
        result = 31 * result + initialPosVariance.hashCode()
        result = 31 * result + initialVelVariance.hashCode()
        result = 31 * result + initialStrideVariance.hashCode()
        result = 31 * result + initialBiasVariance.hashCode()
        result = 31 * result + minStrideMeters.hashCode()
        result = 31 * result + maxStrideMeters.hashCode()
        return result
    }
}

class DiagnosticEkf(val config: EkfConfig = EkfConfig()) {

    companion object {
        const val IDX_X = 0
        const val IDX_Y = 1
        const val IDX_VX = 2
        const val IDX_VY = 3
        const val IDX_STRIDE = 4
        const val IDX_COMPASS_BIAS = 5
    }

    // State vector X = [x, y, vx, vy, L, b_theta]
    private val x = DoubleArray(6)

    // Covariance matrix P (6x6)
    private val p = Array(6) { DoubleArray(6) }

    val stateX: Double get() = x[IDX_X]
    val stateY: Double get() = x[IDX_Y]
    val stateVx: Double get() = x[IDX_VX]
    val stateVy: Double get() = x[IDX_VY]
    val strideLength: Double get() = x[IDX_STRIDE]
    val compassBias: Double get() = x[IDX_COMPASS_BIAS]

    private var isInitializedWithGps = false
    val isInitialized: Boolean
        get() = isInitializedWithGps

    private var lastStepTimestampMs: Long = 0L
    private var currentHeadingRad: Double = 0.0
    private var stepCount: Int = 0
    private var gpsFixCount: Int = 0

    init {
        reset()
    }

    fun reset() {
        x[IDX_X] = 0.0 // x (Irish Grid Easting in meters)
        x[IDX_Y] = 0.0 // y (Irish Grid Northing in meters)
        x[IDX_VX] = 0.0 // vx
        x[IDX_VY] = 0.0 // vy
        x[IDX_STRIDE] = 0.75 // L (stride length in meters)
        x[IDX_COMPASS_BIAS] = 0.0 // b_theta (compass bias in radians)

        for (i in 0..5) {
            for (j in 0..5) {
                p[i][j] = 0.0
            }
        }
        p[IDX_X][IDX_X] = config.initialPosVariance
        p[IDX_Y][IDX_Y] = config.initialPosVariance
        p[IDX_VX][IDX_VX] = config.initialVelVariance
        p[IDX_VY][IDX_VY] = config.initialVelVariance
        p[IDX_STRIDE][IDX_STRIDE] = config.initialStrideVariance
        p[IDX_COMPASS_BIAS][IDX_COMPASS_BIAS] = config.initialBiasVariance

        isInitializedWithGps = false
        lastStepTimestampMs = 0L
        currentHeadingRad = 0.0
        stepCount = 0
        gpsFixCount = 0
    }

    @Synchronized
    fun predict(dtSec: Double = 1.0): EkfState {
        if (!isInitializedWithGps) {
            return getEkfState()
        }

        // State transition: x = x + vx * dt, y = y + vy * dt
        x[IDX_X] += x[IDX_VX] * dtSec
        x[IDX_Y] += x[IDX_VY] * dtSec

        val f = Array(6) { DoubleArray(6) }
        for (i in 0..5) f[i][i] = 1.0
        f[IDX_X][IDX_VX] = dtSec
        f[IDX_Y][IDX_VY] = dtSec

        val fp = Array(6) { DoubleArray(6) }
        for (i in 0..5) {
            for (j in 0..5) {
                var sum = 0.0
                for (k in 0..5) {
                    sum += f[i][k] * p[k][j]
                }
                fp[i][j] = sum
            }
        }

        for (i in 0..5) {
            for (j in 0..5) {
                var sum = 0.0
                for (k in 0..5) {
                    sum += fp[i][k] * f[j][k]
                }
                p[i][j] = sum
            }
            p[i][i] += config.qDiag[i]
        }

        return getEkfState()
    }

    @Synchronized
    fun onStepDetected(headingRawRad: Double, timestampMs: Long) {
        if (!isInitializedWithGps) return

        stepCount++
        currentHeadingRad = headingRawRad
        val bTheta = x[IDX_COMPASS_BIAS]
        val correctedHeading = headingRawRad - bTheta

        val dtStepSec = if (lastStepTimestampMs > 0) {
            (timestampMs - lastStepTimestampMs) / 1000.0
        } else {
            0.6
        }.coerceIn(0.2, 2.0)
        lastStepTimestampMs = timestampMs

        val strideL = x[IDX_STRIDE]
        x[IDX_VX] = (strideL * sin(correctedHeading)) / dtStepSec
        x[IDX_VY] = (strideL * cos(correctedHeading)) / dtStepSec
    }

    @Synchronized
    fun onGpsFix(lat: Double, lon: Double, accuracyMeters: Float, activeIntervalMs: Long): EkfState {
        gpsFixCount++

        val gpsLocation = CoordinateUtils.latLonToMetric(lat, lon)

        if (!isInitializedWithGps) {
            // Initialize EKF state with initial GPS fix
            x[IDX_X] = gpsLocation.x
            x[IDX_Y] = gpsLocation.y
            p[IDX_X][IDX_X] = (accuracyMeters * accuracyMeters).toDouble().coerceAtLeast(1.0)
            p[IDX_Y][IDX_Y] = (accuracyMeters * accuracyMeters).toDouble().coerceAtLeast(1.0)
            isInitializedWithGps = true
            return getEkfState(activeIntervalMs)
        }

        val deltaX = gpsLocation.x - x[IDX_X]
        val deltaY = gpsLocation.y - x[IDX_Y]

        val correctedHeading = currentHeadingRad - x[IDX_COMPASS_BIAS]
        val eLong = deltaX * sin(correctedHeading) + deltaY * cos(correctedHeading)
        val eLat = deltaX * cos(correctedHeading) - deltaY * sin(correctedHeading)

        val rVar = (accuracyMeters * accuracyMeters).toDouble().coerceAtLeast(1.0)

        val hLong = doubleArrayOf(sin(correctedHeading), cos(correctedHeading), 0.0, 0.0, 1.0, 0.0)
        updateScalarObservation(eLong, hLong, rVar)

        val hLat = doubleArrayOf(cos(correctedHeading), -sin(correctedHeading), 0.0, 0.0, 0.0, 1.0)
        updateScalarObservation(eLat, hLat, rVar)

        x[IDX_STRIDE] = x[IDX_STRIDE].coerceIn(config.minStrideMeters, config.maxStrideMeters)
        x[IDX_COMPASS_BIAS] = normalizeAngleRad(x[IDX_COMPASS_BIAS])

        return getEkfState(activeIntervalMs)
    }

    private fun updateScalarObservation(residual: Double, h: DoubleArray, rVar: Double) {
        var s = rVar
        val phT = DoubleArray(6)
        for (i in 0..5) {
            var sum = 0.0
            for (j in 0..5) {
                sum += p[i][j] * h[j]
            }
            phT[i] = sum
            s += h[i] * sum
        }

        if (s <= 1e-9) return

        val k = DoubleArray(6) { phT[it] / s }

        for (i in 0..5) {
            x[i] += k[i] * residual
        }

        val newP = Array(6) { DoubleArray(6) }
        for (i in 0..5) {
            for (j in 0..5) {
                var sum = 0.0
                for (m in 0..5) {
                    val eyeMinusKh = (if (i == m) 1.0 else 0.0) - k[i] * h[m]
                    sum += eyeMinusKh * p[m][j]
                }
                newP[i][j] = sum
            }
        }

        for (i in 0..5) {
            for (j in 0..5) {
                p[i][j] = newP[i][j]
            }
        }
    }

    fun getEkfState(activeIntervalMs: Long = 1000L): EkfState {
        val posXVar = p[IDX_X][IDX_X].coerceAtLeast(1e-4)
        val posYVar = p[IDX_Y][IDX_Y].coerceAtLeast(1e-4)
        val posVarCombined = sqrt(posXVar * posXVar + posYVar * posYVar)

        val velXVar = p[IDX_VX][IDX_VX].coerceAtLeast(1e-4)
        val velYVar = p[IDX_VY][IDX_VY].coerceAtLeast(1e-4)
        val velVarCombined = sqrt(velXVar * velXVar + velYVar * velYVar)

        return EkfState(
            timestampMs = System.currentTimeMillis(),
            position = Estimate(Vector2D(x[IDX_X], x[IDX_Y]), posVarCombined),
            velocity = Estimate(Vector2D(x[IDX_VX], x[IDX_VY]), velVarCombined),
            bearingRad = Estimate(normalizeAngleRad(currentHeadingRad - x[IDX_COMPASS_BIAS]), p[IDX_COMPASS_BIAS][IDX_COMPASS_BIAS]),
            compassBiasRad = Estimate(x[IDX_COMPASS_BIAS], p[IDX_COMPASS_BIAS][IDX_COMPASS_BIAS]),
            strideLengthMeters = Estimate(x[IDX_STRIDE], p[IDX_STRIDE][IDX_STRIDE]),
            stepCount = stepCount,
            gpsFixCount = gpsFixCount,
            activeGpsIntervalMs = activeIntervalMs,
            isInitializedWithGps = isInitializedWithGps
        )
    }

    private fun normalizeAngleRad(angleRad: Double): Double {
        var a = angleRad % (2 * PI)
        if (a < 0) a += 2 * PI
        return a
    }
}

private object IrishGridConverter {
    private const val A = 6377340.189
    private const val B = 6356034.446
    private const val F0 = 1.000035

    private const val LAT0 = 53.5
    private const val LON0 = -8.0
    private const val N0 = 250000.0
    private const val E0 = 200000.0

    private const val WGS84_A = 6378137.0
    private const val WGS84_B = 6356752.3142

    private val gridLetters = arrayOf(
        arrayOf('A', 'B', 'C', 'D', 'E'),
        arrayOf('F', 'G', 'H', 'J', 'K'),
        arrayOf('L', 'M', 'N', 'O', 'P'),
        arrayOf('Q', 'R', 'S', 'T', 'U'),
        arrayOf('V', 'W', 'X', 'Y', 'Z')
    )

    fun Wgs84ToIrishGrid(latLon: LatLon): IrishGridReference {
        val irl75 = wgs84ToIrl75Internal(latLon)
        val rawRef = projectToGridInternal(irl75)

        val correction = getBestOffsetInternal(rawRef.letter, rawRef.easting, rawRef.northing)
        return IrishGridReference(
            letter = rawRef.letter,
            easting = rawRef.easting + correction.eastingOffset,
            northing = rawRef.northing + correction.northingOffset
        )
    }

    fun IrishGridToWgs84(gridRefStr: String): LatLon {
        val clean = gridRefStr.replace(Regex("[^A-Z0-9]"), "").uppercase()
        require(clean.length >= 2) { "Invalid grid reference: $gridRefStr" }
        val letter = clean[0]

        val rawXY = parseGridReferenceToXYInternal(clean)
        val roughLocalE = rawXY.x - getGridLetterEastingOffsetInternal(letter)
        val roughLocalN = rawXY.y % 100000.0

        val correction = getBestOffsetInternal(letter, roughLocalE, roughLocalN)
        val unadjustedX = rawXY.x - correction.eastingOffset
        val unadjustedY = rawXY.y - correction.northingOffset

        val irl75 = inverseProjectToLatLonInternal(unadjustedX, unadjustedY)
        return irl75ToWgs84Internal(irl75)
    }

    private fun getBestOffsetInternal(letter: Char, localEasting: Double, localNorthing: Double): GridOffset {
        val eastBlock = (localEasting / 10000).toInt().coerceIn(0, 9)
        val northBlock = (localNorthing / 10000).toInt().coerceIn(0, 9)
        val cellKey = "$letter-$eastBlock-$northBlock"

        return dynamic10kmOffsetMap[cellKey] ?: letterCorrectionMap[letter] ?: GridOffset(0.0, 0.0)
    }

    private fun getGridLetterEastingOffsetInternal(letter: Char): Double {
        val columns = mapOf(
            'A' to 0, 'F' to 0, 'L' to 0, 'Q' to 0, 'V' to 0,
            'B' to 1, 'G' to 1, 'M' to 1, 'R' to 1, 'W' to 1,
            'C' to 2, 'H' to 2, 'N' to 2, 'S' to 2, 'X' to 2,
            'D' to 3, 'J' to 3, 'O' to 3, 'T' to 3, 'Y' to 3,
            'E' to 4, 'K' to 4, 'P' to 4, 'U' to 4, 'Z' to 4
        )
        return ((columns[letter.uppercaseChar()] ?: 0) * 100000).toDouble()
    }

    private fun parseGridReferenceToXYInternal(cleanRef: String): FullCoordinates {
        val letter = cleanRef[0]
        val numericPart = cleanRef.substring(1)
        val halfLen = numericPart.length / 2

        val localE = numericPart.substring(0, halfLen).padEnd(5, '0').take(5).toDouble()
        val localN = numericPart.substring(halfLen).padEnd(5, '0').take(5).toDouble()

        var baseE = 0.0
        var baseN = 0.0
        outer@ for (r in gridLetters.indices) {
            for (c in gridLetters[r].indices) {
                if (gridLetters[r][c] == letter) {
                    baseE = c * 100000.0
                    baseN = (4 - r) * 100000.0
                    break@outer
                }
            }
        }
        return FullCoordinates(baseE + localE, baseN + localN)
    }

    private fun wgs84ToIrl75Internal(latLon: LatLon): LatLon {
        val lat1 = Math.toRadians(latLon.latitude)
        val lon1 = Math.toRadians(latLon.longitude)
        val eSq1 = (WGS84_A * WGS84_A - WGS84_B * WGS84_B) / (WGS84_A * WGS84_A)
        val nu1 = WGS84_A / sqrt(1.0 - eSq1 * sin(lat1) * sin(lat1))
        val x1 = nu1 * cos(lat1) * cos(lon1)
        val y1 = nu1 * cos(lat1) * sin(lon1)
        val z1 = nu1 * (1.0 - eSq1) * sin(lat1)

        val x2 = (x1 - (-482.530)) / (1.0 - 8.15e-6) + Math.toRadians(0.631 / 3600.0) * y1 - Math.toRadians(0.214 / 3600.0) * z1
        val y2 = (y1 - 130.596) / (1.0 - 8.15e-6) - Math.toRadians(0.631 / 3600.0) * x1 + Math.toRadians(1.042 / 3600.0) * z1
        val z2 = (z1 - (-564.557)) / (1.0 - 8.15e-6) + Math.toRadians(0.214 / 3600.0) * x1 - Math.toRadians(1.042 / 3600.0) * y1

        val eSq2 = (A * A - B * B) / (A * A)
        val p = sqrt(x2 * x2 + y2 * y2)
        var lat2 = atan2(z2, p * (1.0 - eSq2))
        var lon2 = atan2(y2, x2)
        for (i in 0..4) {
            val nu2 = A / sqrt(1.0 - eSq2 * sin(lat2) * sin(lat2))
            lat2 = atan2(z2 + eSq2 * nu2 * sin(lat2), p)
        }
        return LatLon(Math.toDegrees(lat2), Math.toDegrees(lon2))
    }

    private fun irl75ToWgs84Internal(latLon: LatLon): LatLon {
        val lat1 = Math.toRadians(latLon.latitude)
        val lon1 = Math.toRadians(latLon.longitude)
        val eSq1 = (A * A - B * B) / (A * A)
        val nu1 = A / sqrt(1.0 - eSq1 * sin(lat1) * sin(lat1))
        val x1 = nu1 * cos(lat1) * cos(lon1)
        val y1 = nu1 * cos(lat1) * sin(lon1)
        val z1 = nu1 * (1.0 - eSq1) * sin(lat1)

        val x2 = -482.530 + (1.0 + 8.15e-6) * x1 - Math.toRadians(-0.631 / 3600.0) * y1 + Math.toRadians(-0.214 / 3600.0) * z1
        val y2 = 130.596 + Math.toRadians(-0.631 / 3600.0) * x1 + (1.0 + 8.15e-6) * y1 - Math.toRadians(-1.042 / 3600.0) * z1
        val z2 = -564.557 - Math.toRadians(-0.214 / 3600.0) * x1 + Math.toRadians(-1.042 / 3600.0) * y1 + (1.0 + 8.15e-6) * z1

        val eSq2 = (WGS84_A * WGS84_A - WGS84_B * WGS84_B) / (WGS84_A * WGS84_A)
        val p = sqrt(x2 * x2 + y2 * y2)
        var lat2 = atan2(z2, p * (1.0 - eSq2))
        var lon2 = atan2(y2, x2)
        for (i in 0..4) {
            val nu2 = WGS84_A / sqrt(1.0 - eSq2 * sin(lat2) * sin(lat2))
            lat2 = atan2(z2 + eSq2 * nu2 * sin(lat2), p)
        }
        return LatLon(Math.toDegrees(lat2), Math.toDegrees(lon2))
    }

    private fun computeMeridionalArc(lat: Double, lat0: Double, b: Double, n: Double): Double {
        val n2 = n * n
        val n3 = n * n * n
        val ma = (1.0 + n + (5.0 / 4.0) * n2 + (5.0 / 4.0) * n3) * (lat - lat0)
        val mb = (3.0 * n + 3.0 * n2 + (21.0 / 8.0) * n3) * sin(lat - lat0) * cos(lat + lat0)
        val mc = ((15.0 / 8.0) * n2 + (15.0 / 8.0) * n3) * sin(2.0 * (lat - lat0)) * cos(2.0 * (lat + lat0))
        val md = (35.0 / 24.0) * n3 * sin(3.0 * (lat - lat0)) * cos(3.0 * (lat + lat0))
        return b * (ma - mb + mc - md)
    }

    private fun projectToGridInternal(latLon: LatLon): IrishGridReference {
        val latRad = Math.toRadians(latLon.latitude)
        val lonRad = Math.toRadians(latLon.longitude)
        val phi0 = Math.toRadians(LAT0)
        val lambda0 = Math.toRadians(LON0)

        val e2 = (A * A - B * B) / (A * A)
        val n = (A - B) / (A + B)
        val sinLat = sin(latRad)
        val cosLat = cos(latRad)
        val tanLat = tan(latRad)

        val nu = A * F0 / sqrt(1.0 - e2 * sinLat * sinLat)
        val rho = A * F0 * (1.0 - e2) / (1.0 - e2 * sinLat * sinLat).pow(1.5)
        val m = computeMeridionalArc(latRad, phi0, B, n)
        val dLambda = lonRad - lambda0

        val term1 = nu * cosLat * dLambda
        val term2 = nu * cosLat.pow(3) * (nu / rho - tanLat * tanLat) * dLambda.pow(3) / 6.0
        val term3 = nu * cosLat.pow(5) * (5.0 - 18.0 * tanLat * tanLat + tanLat.pow(4)) * dLambda.pow(5) / 120.0
        val eastingTotal = E0 + F0 * (term1 + term2 + term3)

        val nTerm1 = m
        val nTerm2 = nu * sinLat * cosLat * dLambda.pow(2) / 2.0
        val nTerm3 = nu * sinLat * cosLat.pow(3) * (5.0 - tanLat * tanLat + 9.0 * (nu / rho - 1.0)) * dLambda.pow(4) / 24.0
        val northingTotal = N0 + F0 * (nTerm1 + nTerm2 + nTerm3)

        val col = (eastingTotal / 100000.0).toInt().coerceIn(0, 4)
        val row = (4 - (northingTotal / 100000.0).toInt()).coerceIn(0, 4)

        return IrishGridReference(gridLetters[row][col], eastingTotal % 100000.0, northingTotal % 100000.0)
    }

    private fun inverseProjectToLatLonInternal(x: Double, y: Double): LatLon {
        val phi0 = Math.toRadians(LAT0)
        val lambda0 = Math.toRadians(LON0)
        val e2 = (A * A - B * B) / (A * A)
        val n = (A - B) / (A + B)

        var phiPrime = phi0 + (y - N0) / (A * F0)
        var m: Double
        do {
            m = computeMeridionalArc(phiPrime, phi0, B, n)
            phiPrime = (y - N0 - (m * F0)) / (A * F0) + phiPrime
        } while (abs(y - N0 - (m * F0)) >= 0.00001)

        val sinPhi = sin(phiPrime)
        val cosPhi = cos(phiPrime)
        val tanPhi = tan(phiPrime)
        val nuPrime = A * F0 / sqrt(1.0 - e2 * sinPhi * sinPhi)
        val rhoPrime = A * F0 * (1.0 - e2) / (1.0 - e2 * sinPhi * sinPhi).pow(1.5)
        val etaSqPrime = nuPrime / rhoPrime - 1.0

        val xRel = x - E0
        val secPhi = 1.0 / cosPhi
        val vii = tanPhi / (2.0 * rhoPrime * nuPrime * F0 * F0)
        val viii = tanPhi / (24.0 * rhoPrime * nuPrime.pow(3) * F0.pow(4)) * (5.0 + 3.0 * tanPhi * tanPhi + etaSqPrime)

        val lat = phiPrime - vii * xRel.pow(2) + viii * xRel.pow(4)
        val xXi = secPhi / (nuPrime * F0)
        val xXii = secPhi / (6.0 * nuPrime.pow(3) * F0.pow(3)) * (nuPrime / rhoPrime + 2.0 * tanPhi * tanPhi)

        val lon = lambda0 + xXi * xRel - xXii * xRel.pow(3)

        return LatLon(Math.toDegrees(lat), Math.toDegrees(lon))
    }
}

// --- Public CoordinateUtils Object exposing 4 Location-based methods ---

object CoordinateUtils {

    /**
     * Converts WGS84 latitude and longitude into Irish National Grid metric Location (x, y, altitude).
     * Uses 100km subgrid local Easting and Northing (0.0 to 99999.9m) for Alphanumeric Irish Grid compatibility.
     */
    fun latLonToMetric(lat: Double, lon: Double): Location {
        val gridRef = IrishGridConverter.Wgs84ToIrishGrid(LatLon(lat, lon))
        return Location(x = gridRef.easting, y = gridRef.northing, altitude = 0.0)
    }

    /**
     * Converts Irish National Grid metric Location (x, y) into WGS84 latitude and longitude Pair(lat, lon).
     */
    fun metricToLatLon(location: Location): Pair<Double, Double> {
        val letter = 'W'
        val localE = (location.x % 100000.0).let { if (it < 0) it + 100000.0 else it }
        val localN = (location.y % 100000.0).let { if (it < 0) it + 100000.0 else it }
        val gridRefStr = String.format(Locale.US, "%c%05d%05d", letter, localE.toInt(), localN.toInt())
        val latLon = IrishGridConverter.IrishGridToWgs84(gridRefStr)
        return Pair(latLon.latitude, latLon.longitude)
    }

    /**
     * Computes metric Euclidean distance in meters between two Location objects.
     */
    fun getDistance(loc1: Location, loc2: Location): Double {
        return hypot(loc2.x - loc1.x, loc2.y - loc1.y)
    }

    /**
     * Computes grid bearing in degrees (0.0 to 360.0, clockwise from Grid North) from loc1 to loc2.
     */
    fun getBearing(loc1: Location, loc2: Location): Double {
        val dx = loc2.x - loc1.x
        val dy = loc2.y - loc1.y
        var angle = Math.toDegrees(atan2(dx, dy))
        if (angle < 0) {
            angle += 360.0
        }
        return angle
    }

    // --- Private Internal Utility Helpers ---

    private fun parseGridReferenceInternal(ref: String): Location {
        val cleaned = ref.trim().uppercase()
        if (cleaned.isEmpty()) return Location(0.0, 0.0)

        val hasLetter = cleaned[0] in 'A'..'Z'
        val numericPart = if (hasLetter) cleaned.substring(1).replace(Regex("\\s+"), "") else cleaned.replace(Regex("\\s+"), "")

        val halfLen = numericPart.length / 2
        val eastingStr = if (halfLen > 0) numericPart.substring(0, halfLen) else "0"
        val northingStr = if (halfLen > 0) numericPart.substring(halfLen) else numericPart

        val localE = eastingStr.padEnd(5, '0').take(5).toDoubleOrNull() ?: 0.0
        val localN = northingStr.padEnd(5, '0').take(5).toDoubleOrNull() ?: 0.0

        return Location(localE, localN)
    }

    internal fun parseGridRefToLocation(ref: String, altitude: Double = 0.0): Location {
        val loc = parseGridReferenceInternal(ref)
        return Location(loc.x, loc.y, altitude)
    }
}

import android.content.Context

import android.speech.tts.UtteranceProgressListener
import android.util.Log
import java.util.Locale

class TtsManager(context: Context) : TextToSpeech.OnInitListener {

    private var tts: TextToSpeech? = TextToSpeech(context.applicationContext, this)
    private var isInitialized = false
    private var isSpeaking = false

    init {
        tts?.setOnUtteranceProgressListener(object : UtteranceProgressListener() {
            override fun onStart(utteranceId: String?) {
                isSpeaking = true
                Log.d("TtsManager", "TTS Started: $utteranceId")
            }

            override fun onDone(utteranceId: String?) {
                isSpeaking = false
                Log.d("TtsManager", "TTS Done: $utteranceId")
            }

            @Deprecated("Deprecated in Java")
            override fun onError(utteranceId: String?) {
                isSpeaking = false
                Log.e("TtsManager", "TTS Error: $utteranceId")
            }

            override fun onError(utteranceId: String?, errorCode: Int) {
                isSpeaking = false
                Log.e("TtsManager", "TTS Error ($errorCode): $utteranceId")
            }
        })
    }

    override fun onInit(status: Int) {
        if (status == TextToSpeech.SUCCESS) {
            val result = tts?.setLanguage(Locale.UK)
            if (result == TextToSpeech.LANG_MISSING_DATA || result == TextToSpeech.LANG_NOT_SUPPORTED) {
                tts?.setLanguage(Locale.US)
            }
            isInitialized = true
            Log.i("TtsManager", "TextToSpeech initialized successfully.")
        } else {
            Log.e("TtsManager", "TextToSpeech initialization failed with status: $status")
        }
    }

    fun speak(text: String, queueMode: Int = TextToSpeech.QUEUE_ADD) {
        if (!isInitialized || text.isBlank()) return

        val utteranceId = "HikeMsg_${System.currentTimeMillis()}"
        tts?.speak(text, queueMode, null, utteranceId)
    }

    fun stop() {
        if (isInitialized) {
            tts?.stop()
            isSpeaking = false
        }
    }

    fun shutdown() {
        if (isInitialized) {
            tts?.stop()
            tts?.shutdown()
            isInitialized = false
            isSpeaking = false
        }
    }
}

package com.example.pdrandroutetest.util

import android.content.Context
import android.net.Uri
import android.provider.OpenableColumns
import com.example.pdrandroutetest.model.Waypoint
import java.io.BufferedReader
import java.io.StringReader

object RouteParser {
    /**
     * Queries OpenableColumns.DISPLAY_NAME via context.contentResolver when given a content:// URI.
     * Extracts the display name and strips the extension (e.g. "glounthaune.csv" -> "glounthaune").
     */
    fun extractRouteNameFromUri(context: Context, uri: Uri, fallbackName: String? = null): String {
        var displayName: String? = null
        if (uri.scheme == "content") {
            try {
                context.contentResolver.query(
                    uri,
                    arrayOf(OpenableColumns.DISPLAY_NAME),
                    null,
                    null,
                    null
                )?.use { cursor ->
                    if (cursor.moveToFirst()) {
                        val columnIndex = cursor.getColumnIndex(OpenableColumns.DISPLAY_NAME)
                        if (columnIndex != -1) {
                            displayName = cursor.getString(columnIndex)
                        }
                    }
                }
            } catch (_: Exception) {
                // Fallback to uri.lastPathSegment or fallbackName if query fails
            }
        }
        if (displayName.isNullOrBlank()) {
            displayName = fallbackName ?: uri.lastPathSegment
        }
        if (displayName.isNullOrBlank()) {
            displayName = "Imported Route"
        }
        return displayName.substringBeforeLast(".")
    }

    fun extractRouteNameFromUri(context: Context, uriString: String, fallbackName: String? = null): String {
        return try {
            val uri = Uri.parse(uriString)
            extractRouteNameFromUri(context, uri, fallbackName)
        } catch (e: Exception) {
            (fallbackName ?: uriString.substringAfterLast("/")).substringBeforeLast(".")
        }
    }

    /**
     * Parses CSV route content into a list of Waypoints.
     */
    fun parseCsv(csvContent: String): List<Waypoint> {
        val waypoints = mutableListOf<Waypoint>()
        val reader = BufferedReader(StringReader(csvContent))
        val lines = reader.readLines()
        if (lines.isEmpty()) return waypoints

        val headerLine = lines[0].trim()
        val headers = parseCsvLine(headerLine).map { it.lowercase().trim() }

        val nameIdx = headers.indexOfFirst { it in listOf("name", "title") }
        val gridIdx = headers.indexOfFirst { it in listOf("grid_ref", "grid", "location") }
        val latIdx = headers.indexOfFirst { it in listOf("latitude", "lat") }
        val lonIdx = headers.indexOfFirst { it in listOf("longitude", "lon") }
        val altIdx = headers.indexOfFirst { it in listOf("altitude", "alt", "elevation", "ele") }
        val descIdx = headers.indexOfFirst { it in listOf("description", "desc", "waypoint_desc", "waypoint_description", "notes", "comments") }
        val legDescIdx = headers.indexOfFirst { it in listOf("leg_description", "leg_desc", "leg_notes", "leg_comments") }

        for (i in 1 until lines.size) {
            val line = lines[i].trim()
            if (line.isEmpty() || line.startsWith("#")) continue

            val tokens = parseCsvLine(line)
            if (tokens.isEmpty()) continue

            val name = if (nameIdx >= 0 && nameIdx < tokens.size) {
                tokens[nameIdx].trim().removeSurrounding("\"", "\"")
            } else {
                "Waypoint $i"
            }

            val alt = if (altIdx >= 0 && altIdx < tokens.size) {
                tokens[altIdx].trim().removeSurrounding("\"", "\"").toDoubleOrNull() ?: 0.0
            } else {
                0.0
            }

            val description = if (descIdx >= 0 && descIdx < tokens.size) {
                tokens[descIdx].trim().removeSurrounding("\"", "\"")
            } else {
                ""
            }

            val legDescription = if (legDescIdx >= 0 && legDescIdx < tokens.size) {
                tokens[legDescIdx].trim().removeSurrounding("\"", "\"")
            } else {
                ""
            }

            val waypoint = if (gridIdx >= 0 && gridIdx < tokens.size && tokens[gridIdx].isNotBlank()) {
                val gridRef = tokens[gridIdx].trim().removeSurrounding("\"", "\"")
                val loc = CoordinateUtils.parseGridRefToLocation(gridRef, alt)
                Waypoint(
                    x = loc.x,
                    y = loc.y,
                    altitude = alt,
                    name = name,
                    description = description,
                    legDescription = legDescription
                )
            } else if (latIdx >= 0 && lonIdx >= 0 && latIdx < tokens.size && lonIdx < tokens.size) {
                val lat = tokens[latIdx].trim().removeSurrounding("\"", "\"").toDoubleOrNull() ?: 0.0
                val lon = tokens[lonIdx].trim().removeSurrounding("\"", "\"").toDoubleOrNull() ?: 0.0
                Waypoint.fromLatLon(
                    name = name,
                    lat = lat,
                    lon = lon,
                    altitude = alt,
                    description = description,
                    legDescription = legDescription
                )
            } else {
                continue
            }

            waypoints.add(waypoint)
        }

        return waypoints
    }

    fun generateSampleIrishGridCsv(): String {
        return """
            name,grid_ref,altitude,leg_description
            Start,V860870,120.0,Heading north toward Colt
            Colt,V875882,450.0,Ascend ridge toward Summit
            Summit,V890895,850.0,Follow ridge line down
            Ridge,V905910,650.0,Descend to finish point
            Finish,V920925,100.0,Route complete
        """.trimIndent()
    }

    private fun parseCsvLine(line: String): List<String> {
        val tokens = mutableListOf<String>()
        val sb = StringBuilder()
        var inQuotes = false
        for (c in line) {
            when {
                c == '"' -> inQuotes = !inQuotes
                c == ',' && !inQuotes -> {
                    tokens.add(sb.toString())
                    sb.setLength(0)
                }
                else -> sb.append(c)
            }
        }
        tokens.add(sb.toString())
        return tokens
    }
}

package com.example.pdrandroutetest.model

import com.example.pdrandroutetest.util.RouteParser

class Route(
    val name: String,
    val waypoints: List<Waypoint>,
    val legs: List<Leg>
) {
    val totalLengthMeters: Double
        get() = legs.sumOf { it.distanceMeters }

    val numLegs: Int
        get() = legs.size

    companion object {
        fun createFromWaypoints(name: String, rawWaypoints: List<Waypoint>): Route {
            if (rawWaypoints.isEmpty()) {
                return Route(name, emptyList(), emptyList())
            }

            // 1. Process legs sequentially and fill in next pointer & leg descriptions
            val legs = mutableListOf<Leg>()
            for (i in 0 until rawWaypoints.size - 1) {
                val startWp = rawWaypoints[i]
                val targetWp = rawWaypoints[i + 1]
                val legDesc = startWp.legDescription
                val leg = Leg(start = startWp, target = targetWp, legDescription = legDesc)
                legs.add(leg)
            }

            // Wire next pointers
            for (i in 0 until legs.size - 1) {
                legs[i].next = legs[i + 1]
            }

            // Clear legDescription on start waypoints now that legs hold them
            rawWaypoints.forEach { it.legDescription = null }

            // 2. Remove duplicate waypoints by position (matching x and y)
            val uniqueWaypoints = mutableListOf<Waypoint>()
            for (wp in rawWaypoints) {
                val exists = uniqueWaypoints.any { it.x == wp.x && it.y == wp.y }
                if (!exists) {
                    uniqueWaypoints.add(wp)
                }
            }

            return Route(name = name, waypoints = uniqueWaypoints, legs = legs)
        }

        fun parseFromCsvContent(routeName: String, csvContent: String): Route {
            val waypoints = RouteParser.parseCsv(csvContent)
            return createFromWaypoints(routeName, waypoints)
        }
    }
}

package com.example.pdrandroutetest.model

import com.example.pdrandroutetest.util.CoordinateUtils

open class Location(
    open val x: Double = 0.0,
    open val y: Double = 0.0,
    open val altitude: Double = 0.0
) {
    fun getDistance(other: Location): Double {
        return CoordinateUtils.getDistance(this, other)
    }

    fun getBearing(other: Location): Double {
        return CoordinateUtils.getBearing(this, other)
    }

    fun distanceTo(other: Location): Double {
        return getDistance(other)
    }

    fun bearingTo(other: Location): Double {
        return getBearing(other)
    }

    override fun equals(other: Any?): Boolean {
        if (this === other) return true
        if (other !is Location) return false
        return x == other.x && y == other.y && altitude == other.altitude
    }

    override fun hashCode(): Int {
        var result = x.hashCode()
        result = 31 * result + y.hashCode()
        result = 31 * result + altitude.hashCode()
        return result
    }

    override fun toString(): String {
        return "Location(x=$x, y=$y, altitude=$altitude)"
    }
}

package com.example.pdrandroutetest.service

import android.annotation.SuppressLint

import com.example.pdrandroutetest.repository.HikeRepository

class DiagnosticForegroundService : Service() {

    private val serviceScope = CoroutineScope(Dispatchers.Default + SupervisorJob())
    private var tickerJob: Job? = null

    private lateinit var ekfEngine: DiagnosticEkf
    private lateinit var stepDetector: PdrStepDetector
    private lateinit var fusedLocationClient: FusedLocationProviderClient
    private var locationCallback: LocationCallback? = null
    private var ttsManager: TtsManager? = null

    private var wakeLock: PowerManager.WakeLock? = null
    private var breadcrumbIdSeq = 1L
    private var activeGpsIntervalMs = 1000L

    override fun onCreate() {
        super.onCreate()
        ekfEngine = DiagnosticEkf()
        fusedLocationClient = LocationServices.getFusedLocationProviderClient(this)
        ttsManager = TtsManager(this)

        stepDetector = PdrStepDetector(this) { headingRawRad, timestampMs ->
            ekfEngine.onStepDetected(headingRawRad, timestampMs)
            if (HikeRepository.isTracking.value) {
                HikeRepository.telemetryRecorder.recordStep(
                    headingRawRad = headingRawRad,
                    timestampMs = timestampMs,
                    recordedEkfState = ekfEngine.getEkfState()
                )
            }
        }

        val powerManager = getSystemService(Context.POWER_SERVICE) as PowerManager
        wakeLock = powerManager.newWakeLock(PowerManager.PARTIAL_WAKE_LOCK, "PDRWithEKFTest::WakeLock")

        observeGpsPollingInterval()
        observeHikeMessages()
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_START -> startTrackingService()
            ACTION_STOP -> stopTrackingService()
            ACTION_UPDATE_INTERVAL -> {
                val newInterval = intent.getLongExtra(EXTRA_INTERVAL_MS, 1000L)
                updateLocationRequestInterval(newInterval)
            }
        }
        return START_STICKY
    }

    private fun startTrackingService() {
        HikeRepository.clearBreadcrumbs()
        HikeRepository.telemetryRecorder.clear()
        ekfEngine.reset()
        breadcrumbIdSeq = 1L

        createNotificationChannel()
        val notification = buildNotification("PDR Tracking Active (1 Hz EKF)")

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
            startForeground(
                NOTIFICATION_ID,
                notification,
                ServiceInfo.FOREGROUND_SERVICE_TYPE_LOCATION
            )
        } else {
            startForeground(NOTIFICATION_ID, notification)
        }

        wakeLock?.acquire(10 * 60 * 1000L /* 10 minutes timeout */)
        stepDetector.start()
        startLocationUpdates(activeGpsIntervalMs)
        start1HzPredictionTicker()
        HikeRepository.startHike(HikeRepository.activeRoute.value)
        HikeRepository.setTracking(true)
    }

    private fun stopTrackingService() {
        tickerJob?.cancel()
        stepDetector.stop()
        stopLocationUpdates()
        if (wakeLock?.isHeld == true) {
            wakeLock?.release()
        }
        HikeRepository.stopHike()
        HikeRepository.setTracking(false)
        stopForeground(STOP_FOREGROUND_REMOVE)
        stopSelf()
    }

    private fun observeGpsPollingInterval() {
        HikeRepository.gpsPollingIntervalMs
            .onEach { intervalMs ->
                if (intervalMs != activeGpsIntervalMs) {
                    activeGpsIntervalMs = intervalMs
                    updateLocationRequestInterval(intervalMs)
                }
            }
            .launchIn(serviceScope)
    }

    private fun observeHikeMessages() {
        HikeRepository.hikeMessage
            .onEach { message ->
                if (!message.isNullOrBlank()) {
                    ttsManager?.speak(message)
                }
            }
            .launchIn(serviceScope)
    }

    @SuppressLint("MissingPermission")
    private fun startLocationUpdates(intervalMs: Long) {
        stopLocationUpdates()

        val locationRequest = LocationRequest.Builder(Priority.PRIORITY_HIGH_ACCURACY, intervalMs)
            .setMinUpdateIntervalMillis((intervalMs / 2).coerceAtLeast(200L))
            .build()

        locationCallback = object : LocationCallback() {
            override fun onLocationResult(locationResult: LocationResult) {
                for (location in locationResult.locations) {
                    val ekfState = ekfEngine.onGpsFix(
                        lat = location.latitude,
                        lon = location.longitude,
                        accuracyMeters = location.accuracy,
                        activeIntervalMs = activeGpsIntervalMs
                    )
                    if (HikeRepository.isTracking.value) {
                        HikeRepository.telemetryRecorder.recordGpsFix(
                            latitude = location.latitude,
                            longitude = location.longitude,
                            accuracyMeters = location.accuracy,
                            activeIntervalMs = activeGpsIntervalMs,
                            timestampMs = ekfState.timestampMs,
                            recordedEkfState = ekfState
                        )
                    }
                    HikeRepository.updateEkfState(ekfState)

                    serviceScope.launch {
                        val gpsBreadcrumb = Breadcrumb(
                            id = breadcrumbIdSeq++,
                            location = Location(ekfState.position.value.x, ekfState.position.value.y),
                            timestampMs = ekfState.timestampMs,
                            isGpsFix = true,
                            ekfState = ekfState
                        )
                        HikeRepository.emitBreadcrumb(gpsBreadcrumb)
                    }
                }
            }
        }

        try {
            fusedLocationClient.requestLocationUpdates(
                locationRequest,
                locationCallback!!,
                mainLooper
            )
        } catch (e: Exception) {
            e.printStackTrace()
        }
    }

    private fun updateLocationRequestInterval(intervalMs: Long) {
        if (HikeRepository.isTracking.value) {
            startLocationUpdates(intervalMs)
        }
    }

    private fun stopLocationUpdates() {
        locationCallback?.let {
            fusedLocationClient.removeLocationUpdates(it)
            locationCallback = null
        }
    }

    private fun start1HzPredictionTicker() {
        tickerJob?.cancel()
        tickerJob = serviceScope.launch {
            while (isActive) {
                delay(1000L)
                if (ekfEngine.isInitialized) {
                    val ekfState = ekfEngine.predict(1.0)
                    if (HikeRepository.isTracking.value) {
                        HikeRepository.telemetryRecorder.recordTick(
                            dtSec = 1.0,
                            timestampMs = ekfState.timestampMs,
                            recordedEkfState = ekfState
                        )
                    }
                    HikeRepository.updateEkfState(ekfState)

                    val breadcrumb = Breadcrumb(
                        id = breadcrumbIdSeq++,
                        location = Location(ekfState.position.value.x, ekfState.position.value.y),
                        timestampMs = ekfState.timestampMs,
                        isGpsFix = false,
                        ekfState = ekfState
                    )
                    HikeRepository.emitBreadcrumb(breadcrumb)
                }
            }
        }
    }

    private fun createNotificationChannel() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(
                CHANNEL_ID,
                "PDR Diagnostic Tracking",
                NotificationManager.IMPORTANCE_LOW
            ).apply {
                description = "Runs 1 Hz EKF prediction loop and handles sensor fusion."
            }
            val manager = getSystemService(NotificationManager::class.java)
            manager.createNotificationChannel(channel)
        }
    }

    private fun buildNotification(contentText: String): Notification {
        return NotificationCompat.Builder(this, CHANNEL_ID)
            .setContentTitle("PDR Diagnostic Service")
            .setContentText(contentText)
            .setSmallIcon(android.R.drawable.ic_menu_compass)
            .setOngoing(true)
            .setPriority(NotificationCompat.PRIORITY_LOW)
            .build()
    }

    override fun onDestroy() {
        super.onDestroy()
        ttsManager?.shutdown()
        ttsManager = null
    }

    override fun onBind(intent: Intent?): IBinder? = null

    companion object {
        const val CHANNEL_ID = "pdr_diagnostic_channel"
        const val NOTIFICATION_ID = 1001

        const val ACTION_START = "com.example.pdrandroutetest.action.START"
        const val ACTION_STOP = "com.example.pdrandroutetest.action.STOP"
        const val ACTION_UPDATE_INTERVAL = "com.example.pdrandroutetest.action.UPDATE_INTERVAL"
        const val EXTRA_INTERVAL_MS = "com.example.pdrandroutetest.extra.INTERVAL_MS"
    }
}

package com.example.pdrandroutetest.data

import kotlin.math.hypot

data class Vector2D(
    val x: Double = 0.0,
    val y: Double = 0.0
)

data class Estimate<T>(
    val value: T,
    val variance: Double
)

data class EkfState(
    val timestampMs: Long = System.currentTimeMillis(),
    val position: Estimate<Vector2D> = Estimate(Vector2D(0.0, 0.0), 1.0),
    val velocity: Estimate<Vector2D> = Estimate(Vector2D(0.0, 0.0), 0.1),
    val bearingRad: Estimate<Double> = Estimate(0.0, 0.05),
    val compassBiasRad: Estimate<Double> = Estimate(0.0, 0.05),
    val strideLengthMeters: Estimate<Double> = Estimate(0.75, 0.01),
    val stepCount: Int = 0,
    val gpsFixCount: Int = 0,
    val activeGpsIntervalMs: Long = 1000L,
    val isInitializedWithGps: Boolean = false
) {
    val speedMps: Estimate<Double>
        get() {
            val speedVal = hypot(velocity.value.x, velocity.value.y)
            return Estimate(speedVal, velocity.variance)
        }
}

                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Text(
                    text = "6-State EKF Telemetry",
                    style = MaterialTheme.typography.labelMedium,
                    fontWeight = FontWeight.Bold,
                    color = MaterialTheme.colorScheme.primary
                )
                Text(
                    text = if (ekfState.isInitializedWithGps) "GPS Init: OK" else "GPS Init: Pending",
                    style = MaterialTheme.typography.labelSmall,
                    color = if (ekfState.isInitializedWithGps) MaterialTheme.colorScheme.primary else MaterialTheme.colorScheme.error
                )
            }

            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween
            ) {
                // 1. Position: 4 digits X, single space, 4 digits Y (no brackets, no comma, no 'm')
                val x4 = ((ekfState.position.value.x / 10.0).toInt() % 10000).let { if (it < 0) it + 10000 else it }
                val y4 = ((ekfState.position.value.y / 10.0).toInt() % 10000).let { if (it < 0) it + 10000 else it }
                TelemetryItem(
                    label = "Pos (X Y)",
                    value = "%04d %04d".format(x4, y4),
                    errorText = "±%.2f".format(sqrt(ekfState.position.variance))
                )

                // 2. Bearing
                TelemetryItem(
                    label = "Bearing",
                    value = "%.1f°".format(Math.toDegrees(ekfState.bearingRad.value)),
                    errorText = "±%.1f°".format(Math.toDegrees(sqrt(ekfState.bearingRad.variance)))
                )

                // 3. Speed
                TelemetryItem(
                    label = "Speed",
                    value = "%.2fm/s".format(ekfState.speedMps.value),
                    errorText = "±%.2f".format(sqrt(ekfState.speedMps.variance))
                )

                // 4. Step Length
                TelemetryItem(
                    label = "Step L",
                    value = "%.2fm".format(ekfState.strideLengthMeters.value),
                    errorText = "±%.3f".format(sqrt(ekfState.strideLengthMeters.variance))
                )

                // 5. Compass Bias
                TelemetryItem(
                    label = "Bias (b_θ)",
                    value = "%.1f°".format(Math.toDegrees(ekfState.compassBiasRad.value)),
                    errorText = "±%.1f°".format(Math.toDegrees(sqrt(ekfState.compassBiasRad.variance)))
                )
            }
        }
    }
}

internal class ViewportTracker {
    var minX = 0.0
    var maxX = 100.0
    var minY = 0.0
    var maxY = 100.0

    fun updateAll(route: Route?, breadcrumbs: List<Breadcrumb>, activeEkfState: EkfState) {
        var minXVal = Double.MAX_VALUE
        var maxXVal = -Double.MAX_VALUE
        var minYVal = Double.MAX_VALUE
        var maxYVal = -Double.MAX_VALUE
        var hasPoints = false

        route?.waypoints?.forEach { wp ->
            if (wp.x < minXVal) minXVal = wp.x
            if (wp.x > maxXVal) maxXVal = wp.x
            if (wp.y < minYVal) minYVal = wp.y
            if (wp.y > maxYVal) maxYVal = wp.y
            hasPoints = true
        }

        breadcrumbs.forEach { bc ->
            val px = bc.location.x
            val py = bc.location.y
            if ((px != 0.0 || py != 0.0) || route == null) {
                if (px < minXVal) minXVal = px
                if (px > maxXVal) maxXVal = px
                if (py < minYVal) minYVal = py
                if (py > maxYVal) maxYVal = py
                hasPoints = true
            }
        }

        if (activeEkfState.isInitializedWithGps) {
            val ax = activeEkfState.position.value.x
            val ay = activeEkfState.position.value.y
            if (ax < minXVal) minXVal = ax
            if (ax > maxXVal) maxXVal = ax
            if (ay < minYVal) minYVal = ay
            if (ay > maxYVal) maxYVal = ay
            hasPoints = true
        }

        if (!hasPoints) {
            minXVal = 0.0; maxXVal = 100.0; minYVal = 0.0; maxYVal = 100.0
        }

        val spanX = (maxXVal - minXVal).coerceAtLeast(20.0)
        val spanY = (maxYVal - minYVal).coerceAtLeast(20.0)
        val marginX = spanX * 0.15
        val marginY = spanY * 0.15

        minX = minXVal - marginX
        maxX = maxXVal + marginX
        minY = minYVal - marginY
        maxY = maxYVal + marginY
    }
}

@Composable
fun MapScreen(
    route: Route?,
    breadcrumbs: List<Breadcrumb>,
    activeEkfState: EkfState,
    currentTargetWaypoint: Waypoint? = null,
    show1kmGrid: Boolean = false,
    show100mGrid: Boolean = false,
    isLegView: Boolean = false,
    activeLeg: Leg? = null,
    driftCoolOffSec: Long = 2L,
    steadyCoolOffSec: Long = 15L,
    relativeBearingCorrection: Double = 0.0,
    onToggle1kmGrid: (Boolean) -> Unit = {},
    onToggle100mGrid: (Boolean) -> Unit = {},
    onToggleLegView: (Boolean) -> Unit = {},
    onSelectDriftCoolOffSec: (Long) -> Unit = {},
    onSelectSteadyCoolOffSec: (Long) -> Unit = {},
    modifier: Modifier = Modifier
) {
    val viewportTracker = remember { ViewportTracker() }

    var zoomScale by remember { mutableFloatStateOf(1f) }
    var panOffset by remember { mutableStateOf(Offset.Zero) }
    var showSettingsDialog by remember { mutableStateOf(false) }

    val textMeasurer = rememberTextMeasurer()

    val validBreadcrumbs = remember(breadcrumbs, route) {
        breadcrumbs.filter { it.location.x != 0.0 || it.location.y != 0.0 || route == null }
    }

    if (!isLegView) {
        viewportTracker.updateAll(
            route = route,
            breadcrumbs = validBreadcrumbs,
            activeEkfState = activeEkfState
        )
    }

    Box(
        modifier = modifier
            .fillMaxSize()
            .clip(RoundedCornerShape(12.dp))
            .background(Color(0xFF0A0F1A))
            .pointerInput(Unit) {
                detectTransformGestures { _, pan, zoom, _ ->
                    zoomScale = (zoomScale * zoom).coerceIn(0.2f, 15f)
                    panOffset += pan
                }
            }
    ) {
        Canvas(modifier = Modifier.fillMaxSize()) {
            val widthPx = size.width
            val heightPx = size.height

            val spanX = (viewportTracker.maxX - viewportTracker.minX).coerceAtLeast(1.0)
            val spanY = (viewportTracker.maxY - viewportTracker.minY).coerceAtLeast(1.0)

            val scaleX = (widthPx * 0.90f) / spanX.toFloat()
            val scaleY = (heightPx * 0.90f) / spanY.toFloat()
            val baseScale = minOf(scaleX, scaleY)

            val originCx = widthPx / 2f
            val originCy = heightPx / 2f

            val centerX = (viewportTracker.minX + viewportTracker.maxX) / 2.0
            val centerY = (viewportTracker.minY + viewportTracker.maxY) / 2.0

            // Leg View Geometry Setup (start at bottom, target at top)
            val legStart = activeLeg?.start
            val legTarget = activeLeg?.target
            val legDx = if (legStart != null && legTarget != null) legTarget.x - legStart.x else 0.0
            val legDy = if (legStart != null && legTarget != null) legTarget.y - legStart.y else 1.0
            val legLen = hypot(legDx, legDy).coerceAtLeast(1.0)
            val ux = legDx / legLen
            val uy = legDy / legLen
            val vx = uy
            val vy = -ux

            fun toCanvasOffset(xMeters: Double, yMeters: Double): Offset {
                if (isLegView && legStart != null) {
                    val apx = xMeters - legStart.x
                    val apy = yMeters - legStart.y

                    val progress = apx * ux + apy * uy
                    val offset = apx * vx + apy * vy

                    val spanYLeg = legLen * 1.35
                    val spanXLeg = spanYLeg * (widthPx / heightPx).coerceIn(0.5f, 2.0f)
                    val scaleLeg = minOf(widthPx / spanXLeg, heightPx / spanYLeg) * zoomScale

                    val cx = originCx + (offset * scaleLeg).toFloat() + panOffset.x
                    val shiftY = legLen * 0.10
                    val cy = (originCy + ((spanYLeg / 2.0) - progress - shiftY) * scaleLeg + panOffset.y).toFloat()

                    return Offset(cx, cy)
                } else {
                    val rawCx = originCx + ((xMeters - centerX) * baseScale).toFloat()
                    val rawCy = originCy - ((yMeters - centerY) * baseScale).toFloat()

                    val zoomedX = originCx + (rawCx - originCx) * zoomScale + panOffset.x
                    val zoomedY = originCy + (rawCy - originCy) * zoomScale + panOffset.y

                    return Offset(zoomedX, zoomedY)
                }
            }

            // --- Draw Map Grids (only in Map View) ---
            if (!isLegView) {
                val currentScale = baseScale * zoomScale
                if (currentScale > 0f) {
                    val leftMeters = centerX + (0f - originCx - panOffset.x) / currentScale
                    val rightMeters = centerX + (widthPx - originCx - panOffset.x) / currentScale
                    val topMeters = centerY - (0f - originCy - panOffset.y) / currentScale
                    val bottomMeters = centerY - (heightPx - originCy - panOffset.y) / currentScale

                    val minVisX = minOf(leftMeters, rightMeters)
                    val maxVisX = maxOf(leftMeters, rightMeters)
                    val minVisY = minOf(topMeters, bottomMeters)
                    val maxVisY = maxOf(topMeters, bottomMeters)

                    // 1) Draw 100m gridlines (unlabeled)
                    if (show100mGrid) {
                        val spacing100mPx = 100.0 * currentScale.toDouble()
                        if (spacing100mPx >= 4.0) {
                            val start100mX = (kotlin.math.floor(minVisX / 100.0) * 100.0).toInt()
                            val end100mX = (kotlin.math.ceil(maxVisX / 100.0) * 100.0).toInt()

                            for (xMeters in start100mX..end100mX step 100) {
                                if (show1kmGrid && xMeters % 1000 == 0) continue
                                val p = toCanvasOffset(xMeters.toDouble(), minVisY)
                                if (p.x in 0f..widthPx) {
                                    drawLine(
                                        color = Color(0xFF1E293B),
                                        start = Offset(p.x, 0f),
                                        end = Offset(p.x, heightPx),
                                        strokeWidth = 1f
                                    )
                                }
                            }

                            val start100mY = (kotlin.math.floor(minVisY / 100.0) * 100.0).toInt()
                            val end100mY = (kotlin.math.ceil(maxVisY / 100.0) * 100.0).toInt()

                            for (yMeters in start100mY..end100mY step 100) {
                                if (show1kmGrid && yMeters % 1000 == 0) continue
                                val p = toCanvasOffset(minVisX, yMeters.toDouble())
                                if (p.y in 0f..heightPx) {
                                    drawLine(
                                        color = Color(0xFF1E293B),
                                        start = Offset(0f, p.y),
                                        end = Offset(widthPx, p.y),
                                        strokeWidth = 1f
                                    )
                                }
                            }
                        }
                    }

                    // 2) Draw 1km gridlines (labeled by 2-digit Irish Grid reference)
                    if (show1kmGrid) {
                        val start1kmX = (kotlin.math.floor(minVisX / 1000.0) * 1000.0).toInt()
                        val end1kmX = (kotlin.math.ceil(maxVisX / 1000.0) * 1000.0).toInt()

                        for (xMeters in start1kmX..end1kmX step 1000) {
                            val p = toCanvasOffset(xMeters.toDouble(), minVisY)
                            val canvasX = p.x
                            if (canvasX in -20f..(widthPx + 20f)) {
                                drawLine(
                                    color = Color(0xFF334E68).copy(alpha = 0.75f),
                                    start = Offset(canvasX, 0f),
                                    end = Offset(canvasX, heightPx),
                                    strokeWidth = 1.5f
                                )

                                val digits = (kotlin.math.abs(xMeters) / 1000) % 100
                                val labelStr = String.format(Locale.US, "%02d", digits)

                                val textResult = textMeasurer.measure(
                                    text = labelStr,
                                    style = TextStyle(
                                        color = Color(0xFF9FB3C8),
                                        fontSize = 10.sp,
                                        fontWeight = FontWeight.Bold
                                    )
                                )

                                drawText(
                                    textLayoutResult = textResult,
                                    topLeft = Offset(
                                        (canvasX + 4f).coerceIn(4f, widthPx - textResult.size.width - 4f),
                                        8f
                                    )
                                )
                                drawText(
                                    textLayoutResult = textResult,
                                    topLeft = Offset(
                                        (canvasX + 4f).coerceIn(4f, widthPx - textResult.size.width - 4f),
                                        heightPx - textResult.size.height - 8f
                                    )
                                )
                            }
                        }

                        val start1kmY = (kotlin.math.floor(minVisY / 1000.0) * 1000.0).toInt()
                        val end1kmY = (kotlin.math.ceil(maxVisY / 1000.0) * 1000.0).toInt()

                        for (yMeters in start1kmY..end1kmY step 1000) {
                            val p = toCanvasOffset(minVisX, yMeters.toDouble())
                            val canvasY = p.y
                            if (canvasY in -20f..(heightPx + 20f)) {
                                drawLine(
                                    color = Color(0xFF334E68).copy(alpha = 0.75f),
                                    start = Offset(0f, canvasY),
                                    end = Offset(widthPx, canvasY),
                                    strokeWidth = 1.5f
                                )

                                val digits = (kotlin.math.abs(yMeters) / 1000) % 100
                                val labelStr = String.format(Locale.US, "%02d", digits)

                                val textResult = textMeasurer.measure(
                                    text = labelStr,
                                    style = TextStyle(
                                        color = Color(0xFF9FB3C8),
                                        fontSize = 10.sp,
                                        fontWeight = FontWeight.Bold
                                    )
                                )

                                    textLayoutResult = textResult,

                                        8f,

                                    )
                                )
                                drawText(
                                    textLayoutResult = textResult,
                                    topLeft = Offset(
                                        widthPx - textResult.size.width - 8f,
                                        (canvasY - textResult.size.height - 2f).coerceIn(4f, heightPx - textResult.size.height - 4f)
                                    )
                                )
                            }
                        }
                    }
                }
            }

            // Draw current leg arrival & approach circles centered on current leg target (perimeter strokes only, subtle)
            if (activeLeg != null) {
                val targetCenter = toCanvasOffset(activeLeg.target.x, activeLeg.target.y)
                val effectiveScale = if (isLegView) {
                    (minOf(widthPx, heightPx) / (legLen * 1.30)) * zoomScale

                }

                // Approach circle (100m) - perimeter only

                    color = Color(0xFF4CAF50),
                    radius = arriveRadiusPx,

            // Highlight current leg subtly with translucent yellow line under normal leg line
            if (activeLeg != null) {

            // Draw route legs (path lines)

            // Draw route waypoints
            route?.waypoints?.forEach { wp ->

                    val leftY = currentCenter.y - (arrowLengthPx * 1.4f * cos(leftAngle)).toFloat()
                    val rightX = currentCenter.x + (arrowLengthPx * 1.4f * sin(rightAngle)).toFloat()
                    val rightY = currentCenter.y - (arrowLengthPx * 1.4f * cos(rightAngle)).toFloat()

                    lineTo(leftX, leftY)
                    lineTo(rightX, rightY)
                    close()
                }
                drawPath(
                    path = wedgePath,
                    color = Color(0xFFFFD700).copy(alpha = 0.3f)
                )

                val arrowTipX = currentCenter.x + (arrowLengthPx * sin(effectiveBearing)).toFloat()
                val arrowTipY = currentCenter.y - (arrowLengthPx * cos(effectiveBearing)).toFloat()

                drawLine(
                    color = Color(0xFFFFD700),
                    start = currentCenter,
                    end = Offset(arrowTipX, arrowTipY),

                    color = Color(0xFFFFD700),
                    radius = 6f,
                    center = currentCenter

                drawLine(
                    color = Color(0xFFFFB300),
                    start = hikerOffset,
                    end = targetOffset,
                    strokeWidth = 3.5f * zoomScale.coerceIn(0.5f, 2.5f),
                    pathEffect = PathEffect.dashPathEffect(floatArrayOf(12f, 8f), 0f)

        }

        // Status overlay badge
        val validGpsCount = validBreadcrumbs.count { it.isGpsFix }
        Box(
            modifier = Modifier
                .align(Alignment.TopStart)
                .padding(8.dp)

        ) {

                text = "$routeText | BCs: ${validBreadcrumbs.size} (GPS: $validGpsCount)",

            )

        // Upper Right UI Stack (North Indicator + Relative Bearing Indicator)
        if (activeEkfState.isInitializedWithGps) {
            Column(
                modifier = Modifier
                    .align(Alignment.TopEnd)
                    .padding(8.dp),
                verticalArrangement = Arrangement.spacedBy(6.dp),
                horizontalAlignment = Alignment.End
            ) {
                // 1. North Indicator Compass Needle
                Box(
                    modifier = Modifier
                        .background(Color.Black.copy(alpha = 0.75f), RoundedCornerShape(8.dp))
                        .padding(horizontal = 8.dp, vertical = 4.dp),
                    contentAlignment = Alignment.Center
                ) {
                    Row(
                        horizontalArrangement = Arrangement.spacedBy(6.dp),
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        Canvas(modifier = Modifier.size(20.dp)) {

                            val radius = size.width / 2f - 2f
                            val northAngle = if (isLegView && activeLeg != null) {

                            } else {
                                0.0
                            }
                            val angleRad = Math.toRadians(northAngle)
                            val tipX = center.x + (radius * 0.8f * sin(angleRad)).toFloat()
                            val tipY = center.y - (radius * 0.8f * cos(angleRad)).toFloat()

                            drawCircle(color = Color(0xFF1E293B), radius = radius, center = center)
                            drawLine(
                                color = Color(0xFFFF5252),

                            text = "North",

                        .background(Color.Black.copy(alpha = 0.75f), RoundedCornerShape(8.dp))

                ) {
                    Row(
                        horizontalArrangement = Arrangement.spacedBy(6.dp),
                        verticalAlignment = Alignment.CenterVertically

                            Text("Off-Course Drift Cool-off:", style = MaterialTheme.typography.bodyMedium, fontWeight = FontWeight.SemiBold)

                        }

                        Column(verticalArrangement = Arrangement.spacedBy(4.dp)) {
                            Text("Off-Course Steady Cool-off:", style = MaterialTheme.typography.bodyMedium, fontWeight = FontWeight.SemiBold)
                            Text("Cool-off when holding steady off-course heading (seconds)", style = MaterialTheme.typography.labelSmall, color = MaterialTheme.colorScheme.onSurfaceVariant)
                            Row(horizontalArrangement = Arrangement.spacedBy(6.dp)) {

                confirmButton = {

                    }
                }
            )
        }
    }
}

package com.example.pdrandroutetest.repository

import com.example.pdrandroutetest.data.EkfState
import com.example.pdrandroutetest.hike.Hike
import com.example.pdrandroutetest.hike.NavigationMessage
import com.example.pdrandroutetest.model.Breadcrumb
import com.example.pdrandroutetest.model.Leg
import com.example.pdrandroutetest.model.Route
import com.example.pdrandroutetest.model.Waypoint
import com.example.pdrandroutetest.telemetry.TelemetryPlaybackEngine
import com.example.pdrandroutetest.telemetry.TelemetryRecorder
import kotlinx.coroutines.ExperimentalCoroutinesApi
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import java.util.concurrent.ConcurrentLinkedQueue

object HikeRepository {
    private val _ekfStateFlow = MutableStateFlow(EkfState())
    val ekfStateFlow: StateFlow<EkfState> = _ekfStateFlow.asStateFlow()

    private val _breadcrumbsFlow = MutableSharedFlow<Breadcrumb>(replay = 500)
    val breadcrumbsFlow: SharedFlow<Breadcrumb> = _breadcrumbsFlow.asSharedFlow()

    private val _isTracking = MutableStateFlow(false)
    val isTracking: StateFlow<Boolean> = _isTracking.asStateFlow()

    private val _gpsPollingIntervalMs = MutableStateFlow(1000L)
    val gpsPollingIntervalMs: StateFlow<Long> = _gpsPollingIntervalMs.asStateFlow()

    private val _activeRoute = MutableStateFlow<Route?>(null)
    val activeRoute: StateFlow<Route?> = _activeRoute.asStateFlow()

    private val _show1kmGrid = MutableStateFlow(false)
    val show1kmGrid: StateFlow<Boolean> = _show1kmGrid.asStateFlow()

    private val _show100mGrid = MutableStateFlow(false)
    val show100mGrid: StateFlow<Boolean> = _show100mGrid.asStateFlow()

    private val _isLegView = MutableStateFlow(false)
    val isLegView: StateFlow<Boolean> = _isLegView.asStateFlow()

    private val _driftCoolOffSec = MutableStateFlow(2L)
    val driftCoolOffSec: StateFlow<Long> = _driftCoolOffSec.asStateFlow()

    private val _steadyCoolOffSec = MutableStateFlow(15L)
    val steadyCoolOffSec: StateFlow<Long> = _steadyCoolOffSec.asStateFlow()

    private val _hikeMessage = MutableStateFlow<String?>(null)
    val hikeMessage: StateFlow<String?> = _hikeMessage.asStateFlow()

    private val _navigationMessages = MutableStateFlow<List<NavigationMessage>>(emptyList())
    val navigationMessages: StateFlow<List<NavigationMessage>> = _navigationMessages.asStateFlow()

    private val _relativeBearingCorrection = MutableStateFlow(0.0)
    val relativeBearingCorrection: StateFlow<Double> = _relativeBearingCorrection.asStateFlow()

    private val _currentTargetWaypoint = MutableStateFlow<Waypoint?>(null)
    val currentTargetWaypoint: StateFlow<Waypoint?> = _currentTargetWaypoint.asStateFlow()

    private val _activeLeg = MutableStateFlow<Leg?>(null)
    val activeLeg: StateFlow<Leg?> = _activeLeg.asStateFlow()

    private var currentHike: Hike? = null
    val activeHike: Hike?
        get() = currentHike

    val telemetryRecorder = TelemetryRecorder()
    val telemetryPlaybackEngine = TelemetryPlaybackEngine()

    private val breadcrumbHistory = ConcurrentLinkedQueue<Breadcrumb>()

    fun updateEkfState(state: EkfState) {
        _ekfStateFlow.value = state
    }

    suspend fun emitBreadcrumb(breadcrumb: Breadcrumb) {
        breadcrumbHistory.add(breadcrumb)
        _breadcrumbsFlow.emit(breadcrumb)
        currentHike?.let { hike ->
            val msg = hike.update(
                location = breadcrumb.location,
                timestampMs = breadcrumb.timestampMs,
                isGpsFix = breadcrumb.isGpsFix,
                ekfState = breadcrumb.ekfState
            )
            if (!msg.isNullOrBlank()) {
                _hikeMessage.value = msg
                _navigationMessages.value = _navigationMessages.value + NavigationMessage(
                    breadcrumbIndex = breadcrumbHistory.size,
                    text = msg,
                    timestampMs = breadcrumb.timestampMs
                )
            }
            _relativeBearingCorrection.value = hike.relativeBearingCorrection(breadcrumb.location, breadcrumb.ekfState)
        }
        _currentTargetWaypoint.value = currentHike?.currentLeg?.target ?: _activeRoute.value?.legs?.firstOrNull()?.target
        _activeLeg.value = currentHike?.currentLeg ?: _activeRoute.value?.legs?.firstOrNull()
    }

    fun startHike(route: Route?) {
        currentHike = Hike(route).apply {
            driftCoolOffSec = _driftCoolOffSec.value
            steadyCoolOffSec = _steadyCoolOffSec.value
        }
        _hikeMessage.value = null
        _navigationMessages.value = emptyList()
        _relativeBearingCorrection.value = 0.0
        _currentTargetWaypoint.value = currentHike?.currentLeg?.target ?: route?.legs?.firstOrNull()?.target
        _activeLeg.value = currentHike?.currentLeg ?: route?.legs?.firstOrNull()
        clearBreadcrumbs()
    }

    fun stopHike() {
        currentHike?.stop()
    }

    fun setTracking(tracking: Boolean) {
        _isTracking.value = tracking
    }

    @OptIn(ExperimentalCoroutinesApi::class)
    fun clearBreadcrumbs() {
        breadcrumbHistory.clear()
        _breadcrumbsFlow.resetReplayCache()
    }

    fun setGpsPollingInterval(intervalMs: Long) {
        _gpsPollingIntervalMs.value = intervalMs
    }

    fun setActiveRoute(route: Route?) {
        _activeRoute.value = route
        _currentTargetWaypoint.value = currentHike?.currentLeg?.target ?: route?.legs?.firstOrNull()?.target
        _activeLeg.value = currentHike?.currentLeg ?: route?.legs?.firstOrNull()
    }

    fun setShow1kmGrid(show: Boolean) {
        _show1kmGrid.value = show
    }

    fun setShow100mGrid(show: Boolean) {
        _show100mGrid.value = show
    }

    fun setLegView(isLeg: Boolean) {
        _isLegView.value = isLeg
    }

    fun setDriftCoolOffSec(sec: Long) {
        _driftCoolOffSec.value = sec.coerceAtLeast(1L)
        currentHike?.driftCoolOffSec = sec.coerceAtLeast(1L)
    }

    fun setSteadyCoolOffSec(sec: Long) {
        _steadyCoolOffSec.value = sec.coerceAtLeast(1L)
        currentHike?.steadyCoolOffSec = sec.coerceAtLeast(1L)
    }

    fun getBreadcrumbHistory(): List<Breadcrumb> {
        return breadcrumbHistory.toList()
    }
}
